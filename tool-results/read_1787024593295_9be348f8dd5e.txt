    61→// ─── Format helpers ───
    62→function formatTimeShort(seconds: number): string {
    63→  const m = Math.floor(seconds / 60);
    64→  const s = seconds % 60;
    65→  return `${m}:${s.toString().padStart(2, '0')}`;
    66→}
    67→
    68→function formatRelativeDate(dateStr: string): string {
    69→  const diff = Date.now() - new Date(dateStr).getTime();
    70→  const min = Math.floor(diff / 60000);
    71→  if (min < 1) return "À l'instant";
    72→  if (min < 60) return `Il y a ${min}min`;
    73→  const h = Math.floor(min / 60);
    74→  if (h < 24) return `Il y a ${h}h`;
    75→  const d = Math.floor(h / 24);
    76→  if (d < 7) return `Il y a ${d}j`;
    77→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' });
    78→}

    80→// ─── Segment generator for movies ───
    81→function generateMovieSegments(runtime: number): { index: number; title: string; start: number }[] {
    82→  if (!runtime || runtime <= 0) return [{ index: 0, title: 'Début', start: 0 }];
    83→  const segmentLength = 5;
    84→  const segments = [];
    85→  for (let i = 0; i < Math.ceil(runtime / segmentLength); i++) {
    86→    const startMin = i * segmentLength;
    87→    const endMin = Math.min((i + 1) * segmentLength, runtime);
    88→    segments.push({
    89→      index: i,
    90→      title: `${formatTimeShort(startMin * 60)} - ${formatTimeShort(endMin * 60)}`,
    91→      start: startMin * 60,
    92→    });
    93→  }
    94→  return segments;
    95→}

   108→// ─── Smart Short Player — compact version for shorts with auto-fallback ───
   109→function SmartShortPlayer({ src, title, onSwitchServer, serverIndex, totalServers }: {
   110→  src: string; title: string; onSwitchServer: () => void; serverIndex: number; totalServers: number;
   111→}) {
   112→  const [state, setState] = useState<'loading' | 'ready' | 'error' | 'timeout'>('loading');
   113→  const [failedServers, setFailedServers] = useState<Set<number>>(new Set());
   114→  const [autoFallbackAttempted, setAutoFallbackAttempted] = useState(false);
   115→  const [allFailed, setAllFailed] = useState(false);
   116→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   117→
   118→  useEffect(() => {
   119→    setState('loading');
   120→    setAutoFallbackAttempted(false);
   121→    if (timerRef.current) clearTimeout(timerRef.current);
   122→    timerRef.current = setTimeout(() => {
   123→      setState(prev => prev === 'loading' ? 'timeout' : prev);
   124→      setFailedServers(prev => new Set(prev).add(serverIndex));
   125→    }, 15000);
   126→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   127→  }, [src, serverIndex]);
   128→
   129→  // Auto-fallback when server fails
   130→  useEffect(() => {
   131→    if ((state === 'error' || state === 'timeout') && !autoFallbackAttempted) {
   132→      setAutoFallbackAttempted(true);
   133→      const newFailed = new Set(failedServers);
   134→      newFailed.add(serverIndex);
   135→      if (newFailed.size >= totalServers) {
   136→        setAllFailed(true);
   137→        return;
   138→      }
   139→      const timer = setTimeout(() => onSwitchServer(), 1500);
   140→      return () => clearTimeout(timer);
   141→    }
   142→  }, [state, autoFallbackAttempted, failedServers, serverIndex, totalServers, onSwitchServer]);
   143→
   144→  const handleLoad = useCallback(() => {
   145→    if (timerRef.current) clearTimeout(timerRef.current);
   146→    setState('ready');
   147→    setFailedServers(prev => { const n = new Set(prev); n.delete(serverIndex); return n; });
   148→  }, [serverIndex]);
   149→
   150→  const handleError = useCallback(() => {
   151→    if (timerRef.current) clearTimeout(timerRef.current);
   152→    setState('error');
   153→    setFailedServers(prev => new Set(prev).add(serverIndex));
   154→  }, [serverIndex]);
   155→
   156→  const showOverlay = state === 'error' || state === 'timeout';
   157→  const serverName = Object.values(API_CONFIG.videoServers)[serverIndex]?.name || `Serveur ${serverIndex + 1}`;
   158→  const available = totalServers - failedServers.size;
   159→  const isAutoFallingBack = showOverlay && !allFailed && !autoFallbackAttempted;
   160→
   161→  return (
   162→    <div className="absolute inset-0 flex items-center justify-center">
   163→      <iframe
   164→        src={src}
   165→        className={cn('w-full h-full transition-opacity duration-500', showOverlay ? 'opacity-0' : 'opacity-100')}
   166→        style={{ aspectRatio: '16/9', minHeight: '100%', maxWidth: '177.78%' }}
   167→        allowFullScreen
   168→        allow="autoplay; fullscreen; picture-in-picture"
   169→        referrerPolicy="origin"
   170→        title={title}
   171→        onLoad={handleLoad}
   172→        onError={handleError}
   173→      />
   174→      {state === 'loading' && !showOverlay && (
   175→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-black/60">
   176→          <div className="flex flex-col items-center gap-2">
   177→            <div className="w-6 h-6 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   178→            <span className="text-[10px] text-white/30">{serverName}</span>
   179→          </div>
   180→        </div>
   181→      )}
   182→      {/* Auto-fallback transition */}
   183→      {isAutoFallingBack && (
   184→        <div className="absolute inset-0 flex items-center justify-center z-15 bg-black/80">
   185→          <div className="flex flex-col items-center gap-2 animate-pulse">
   186→            <Wifi className="w-6 h-6 text-primary" />
   187→            <p className="text-[11px] text-white/50">Serveur suivant...</p>
   188→          </div>
   189→        </div>
   190→      )}
   191→      {/* All servers failed */}
   192→      {showOverlay && allFailed && (
   193→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   194→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   195→            <WifiOff className="w-8 h-8 text-red-400" />
   196→            <p className="text-xs font-bold text-white">Contenu indisponible</p>
   197→            <p className="text-[10px] text-white/30">Aucun serveur ne peut charger ce contenu. Réessayez plus tard.</p>
   198→            <button onClick={() => { setAllFailed(false); setFailedServers(new Set()); onSwitchServer(); }}
   199→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   200→              <RefreshCw className="w-3.5 h-3.5" />
   201→              Réessayer
   202→            </button>
   203→          </div>
   204→        </div>
   205→      )}
   206→      {/* Some servers still available */}
   207→      {showOverlay && !allFailed && !isAutoFallingBack && (
   208→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   209→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   210→            <AlertTriangle className="w-8 h-8 text-amber-400" />
   211→            <p className="text-xs font-bold text-white">Serveur indisponible</p>
   212→            <p className="text-[10px] text-white/30">{serverName} ne répond pas</p>
   213→            <p className="text-[10px] text-primary/50">{available} serveur{available > 1 ? 's' : ''} restant{available > 1 ? 's' : ''}</p>
   214→            <button onClick={onSwitchServer}
   215→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   216→              <SkipForward className="w-3.5 h-3.5" />
   217→              Serveur suivant
   218→            </button>
   219→            <p className="text-[9px] text-white/15">{serverIndex + 1} / {totalServers} · {failedServers.size} échoué{failedServers.size > 1 ? 's' : ''}</p>
   220→          </div>
   221→        </div>
   222→      )}
   223→    </div>
   224→  );
   225→}

   237→// ─── Load shorts from a single endpoint, deduped ───
   238→async function loadShortsFromEndpoint(
   239→  endpoint: string,
   240→  mediaType: 'movie' | 'tv',
   241→  genreLabel?: string,
   242→  maxItems: number = 8,
   243→  seenIds: Set<string> = new Set()
   244→): Promise<ShortItem[]> {
   245→  const data = await fetchTMDB<TMDBResponse<Media>>(endpoint);
   246→  if (!data?.results) return [];
   247→
   248→  const items: ShortItem[] = [];
   249→
   250→  for (const item of data.results) {
   251→    if (items.length >= maxItems) break;
   252→
   253→    // Dedup key: same id + same mediaType = skip
   254→    const dedupKey = `${item.id}-${mediaType}`;
   255→    if (seenIds.has(dedupKey)) continue;
   256→    seenIds.add(dedupKey);
   257→
   258→    if (mediaType === 'tv') {
   259→      // For TV: fetch first episode
   260→      const detail = await fetchTMDB<any>(`/tv/${item.id}`);
   261→      if (detail?.seasons) {
   262→        const firstRealSeason = detail.seasons.find((s: any) => s.season_number > 0);
   263→        if (firstRealSeason) {
   264→          const seasonData = await fetchTMDB<any>(`/tv/${item.id}/season/${firstRealSeason.season_number}`);
   265→          if (seasonData?.episodes?.[0]) {
   266→            const ep = seasonData.episodes[0];
   267→            items.push({
   268→              id: item.id,
   269→              mediaType: 'tv',
   270→              title: item.name || item.original_name || '',
   271→              overview: ep.overview || item.overview || '',
   272→              posterPath: item.poster_path,
   273→              backdropPath: ep.still_path || item.backdrop_path,
   274→              voteAverage: item.vote_average,
   275→              genreLabel,
   276→              season: firstRealSeason.season_number,
   277→              episode: ep.episode_number,
   278→              episodeName: ep.name,
   279→            });
   280→          }
   281→        }
   282→      }
   283→    } else {
   284→      // For movies: single card (no segment splitting for cleaner feed)
   285→      items.push({
   286→        id: item.id,
   287→        mediaType: 'movie',
   288→        title: item.title || item.original_title || '',
   289→        overview: item.overview || '',
   290→        posterPath: item.poster_path,
   291→        backdropPath: item.backdrop_path,
   292→        voteAverage: item.vote_average,
   293→        genreLabel,
   294→      });
   295→    }
   296→  }
   297→
   298→  return items;
   299→}

   301→// ─── Like Button with TikTok-style animation ───
   302→function LikeButton({ mediaId, mediaType, segment, season, likeTriggerRef }: { mediaId: number; mediaType: string; segment: number; season?: number | null; likeTriggerRef?: React.MutableRefObject<(() => void) | null> }) {
   303→  const [data, setData] = useState<LikeData>({ count: 0, liked: false });
   304→  const [animating, setAnimating] = useState(false);
   305→  const [justLiked, setJustLiked] = useState(false); // track if we JUST liked for animation direction
   306→  const [isToggling, setIsToggling] = useState(false); // prevent double-clicks
   307→  const { data: session } = useSession();
   308→  const normalizedSeason = season || 0;
   309→
   310→  useEffect(() => {
   311→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   312→    fetch(`/api/shorts/like?${params}`)
   313→      .then(r => r.ok ? r.json() : null)
   314→      .then(d => { if (d?.count !== undefined) setData(d); })
   315→      .catch(() => {});
   316→  }, [mediaId, mediaType, segment, normalizedSeason]);
   317→
   318→  const toggleLike = async () => {
   319→    if (!session || isToggling) return;
   320→    setIsToggling(true);
   321→    
   322→    // Optimistic update — immediately show the animation
   323→    const willLike = !data.liked;
   324→    setJustLiked(willLike);
   325→    setAnimating(true);
   326→    setData(prev => ({ count: prev.count + (willLike ? 1 : -1), liked: willLike }));
   327→
   328→    try {
   329→      const res = await fetch('/api/shorts/like', {
   330→        method: 'POST',
   331→        headers: { 'Content-Type': 'application/json' },
   332→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason }),
   333→      });
   334→      
   335→      if (!res.ok) {
   336→        // HTTP error (401, 500, etc.) — revert optimistic update
   337→        console.warn('[like] HTTP error:', res.status);
   338→        setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   339→      } else {
   340→        const d = await res.json();
   341→        if (d?.liked !== undefined) {
   342→          // Server confirmed — use server state
   343→          setData(prev => ({ count: d.count ?? prev.count, liked: d.liked }));
   344→        } else if (d?.error) {
   345→          // Server error — revert optimistic update
   346→          console.warn('[like] API error:', d.error);
   347→          setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   348→        }
   349→      }
   350→    } catch (err) {
   351→      // Network error — revert optimistic update
   352→      console.warn('[like] Network error:', err);
   353→      setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   354→    }
   355→    setTimeout(() => { setAnimating(false); setJustLiked(false); setIsToggling(false); }, 600);
   356→  };
   357→
   358→  // Expose toggleLike via ref so parent (double-tap) can trigger it
   359→  useEffect(() => {
   360→    if (likeTriggerRef) likeTriggerRef.current = toggleLike;
   361→  }); // update every render to keep ref fresh
   362→
   363→  return (
   364→    <button onClick={toggleLike} className="flex flex-col items-center gap-0.5 group active:scale-90 transition-transform">
   365→      <div className="relative">
   366→        <Heart
   367→          className={cn(
   368→            'w-7 h-7 transition-all duration-300',
   369→            animating && justLiked && 'scale-125',
   370→            data.liked
   371→              ? 'text-red-500 fill-red-500 drop-shadow-[0_0_8px_rgba(239,68,68,0.4)]'
   372→              : 'text-white/80 group-hover:text-white'
   373→          )}
   374→          strokeWidth={data.liked ? 0 : 1.5}
   375→        />
   376→        {animating && (
   377→          <div className="absolute inset-0 flex items-center justify-center">
   378→            <div className={cn(
   379→              'w-12 h-12 rounded-full animate-ping-once',
   380→              justLiked ? 'bg-red-500/25' : 'bg-white/10'
   381→            )} />
   382→          </div>
   383→        )}
   384→      </div>
   385→      <span className={cn('text-[11px] font-medium transition-colors', data.liked ? 'text-red-400' : 'text-white/40')}>
   386→        {data.count > 0 ? data.count : 'J\'aime'}
   387→      </span>
   388→    </button>
   389→  );
   390→}

   393→function CommentSheet({ mediaId, mediaType, segment, season, onClose }: {
   394→  mediaId: number; mediaType: string; segment: number; season?: number | null; onClose: () => void;
   395→}) {
   396→  const [comments, setComments] = useState<CommentData[]>([]);
   397→  const [newComment, setNewComment] = useState('');
   398→  const [submitting, setSubmitting] = useState(false);
   399→  const [visible, setVisible] = useState(false);
   400→  const { data: session } = useSession();
   401→  const normalizedSeason = season || 0;
   402→
   403→  useEffect(() => { requestAnimationFrame(() => setVisible(true)); }, []);
   404→
   405→  useEffect(() => {
   406→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   407→    fetch(`/api/shorts/comments?${params}`)
   408→      .then(r => r?.json())
   409→      .then(d => { if (d?.comments) setComments(d.comments); })
   410→      .catch(() => {});
   411→  }, [mediaId, mediaType, segment, normalizedSeason]);
   412→
   413→  const handleClose = () => { setVisible(false); setTimeout(onClose, 250); };
   414→  const handleBackdropClick = (e: React.MouseEvent) => { if (e.target === e.currentTarget) handleClose(); };
   415→
   416→  const submitComment = async () => {
   417→    if (!session || !newComment.trim() || submitting) return;
   418→    setSubmitting(true);
   419→    try {
   420→      const res = await fetch('/api/shorts/comments', {
   421→        method: 'POST',
   422→        headers: { 'Content-Type': 'application/json' },
   423→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason, content: newComment.trim() }),
   424→      });
   425→      const d = await res.json();
   426→      if (d?.comment) { setComments(prev => [d.comment, ...prev]); setNewComment(''); }
   427→    } catch {}
   428→    setSubmitting(false);
   429→  };
   430→
   431→  return (
   432→    <div className="absolute inset-0 z-50 flex flex-col justify-end" onClick={handleBackdropClick}>
   433→      <div className={cn('absolute inset-0 bg-black/40 transition-opacity duration-250', visible ? 'opacity-100' : 'opacity-0')} />
   434→      <div className={cn(
   435→        'relative z-10 flex flex-col rounded-t-2xl transition-transform duration-250 ease-out',
   436→        'bg-[#1a1a1a]/95 backdrop-blur-xl border-t border-white/[0.06]',
   437→        visible ? 'translate-y-0' : 'translate-y-full'
   438→      )} style={{ height: '55vh', maxHeight: '55dvh' }} onClick={e => e.stopPropagation()}>
   439→        <div className="flex justify-center pt-2.5 pb-1"><div className="w-8 h-1 rounded-full bg-white/15" /></div>
   440→        <div className="flex items-center justify-between px-4 pb-2.5 border-b border-white/[0.06]">
   441→          <h3 className="text-[13px] font-bold text-white/90">Commentaires<span className="ml-1.5 text-white/30 font-normal">{comments.length}</span></h3>
   442→          <button onClick={handleClose} className="p-1.5 rounded-full hover:bg-white/5 transition-colors"><X className="w-4 h-4 text-white/40" /></button>
   443→        </div>
   444→        <div className="flex-1 overflow-y-auto px-4 py-3 space-y-3 scrollbar-hide">
   445→          {comments.length === 0 ? (
   446→            <div className="flex flex-col items-center justify-center py-8">
   447→              <MessageCircle className="w-7 h-7 text-white/[0.06] mb-2" />
   448→              <p className="text-xs text-white/25">Aucun commentaire</p>
   449→              <p className="text-[10px] text-white/15 mt-0.5">Soyez le premier !</p>
   450→            </div>
   451→          ) : comments.map(c => (
   452→            <div key={c.id} className="flex gap-2.5">
   453→              <div className="flex-shrink-0 w-7 h-7 rounded-full bg-gradient-to-br from-primary/30 to-amber-500/20 flex items-center justify-center">
   454→                <span className="text-[9px] font-bold text-primary/80">{(c.user.name || c.user.email)[0].toUpperCase()}</span>
   455→              </div>
   456→              <div className="flex-1 min-w-0">
   457→                <div className="flex items-center gap-2">
   458→                  <span className="text-[11px] font-semibold text-white/70">{c.user.name || 'Anonyme'}</span>
   459→                  <span className="text-[9px] text-white/15">{formatRelativeDate(c.createdAt)}</span>
   460→                </div>
   461→                <p className="text-[12px] text-white/50 mt-0.5 leading-relaxed">{c.content}</p>
   462→              </div>
   463→            </div>
   464→          ))}
   465→        </div>
   466→        {session ? (
