    19→// ─── Fetch helper ───
    20→const fetchTMDB = async <T,>(endpoint: string, timeoutMs: number = 10000): Promise<T | null> => {
    21→  try {
    22→    const controller = new AbortController();
    23→    const tid = setTimeout(() => controller.abort(), timeoutMs);
    24→    const res = await fetch(
    25→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`,
    26→      { signal: controller.signal }
    27→    );
    28→    clearTimeout(tid);
    29→    return res.ok ? await res.json() : null;
    30→  } catch { return null; }
    31→};

    97→// ─── Format helpers ───
    98→function formatTimeShort(seconds: number): string {
    99→  const m = Math.floor(seconds / 60);
   100→  const s = seconds % 60;
   101→  return `${m}:${s.toString().padStart(2, '0')}`;
   102→}

   104→function formatRelativeDate(dateStr: string): string {
   105→  const diff = Date.now() - new Date(dateStr).getTime();
   106→  const min = Math.floor(diff / 60000);
   107→  if (min < 1) return "À l'instant";
   108→  if (min < 60) return `Il y a ${min}min`;
   109→  const h = Math.floor(min / 60);
   110→  if (h < 24) return `Il y a ${h}h`;
   111→  const d = Math.floor(h / 24);
   112→  if (d < 7) return `Il y a ${d}j`;
   113→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' });
   114→}

   116→// ─── Segment generator for movies ───
   117→function generateMovieSegments(runtime: number): { index: number; title: string; start: number }[] {
   118→  if (!runtime || runtime <= 0) return [{ index: 0, title: 'Début', start: 0 }];
   119→  const segmentLength = 5;
   120→  const segments = [];
   121→  for (let i = 0; i < Math.ceil(runtime / segmentLength); i++) {
   122→    const startMin = i * segmentLength;
   123→    const endMin = Math.min((i + 1) * segmentLength, runtime);
   124→    segments.push({
   125→      index: i,
   126→      title: `${formatTimeShort(startMin * 60)} - ${formatTimeShort(endMin * 60)}`,
   127→      start: startMin * 60,
   128→    });
   129→  }
   130→  return segments;
   131→}

   144→// ─── Smart Short Player — compact version for shorts with auto-fallback ───
   145→function SmartShortPlayer({ src, title, onSwitchServer, serverIndex, totalServers }: {
   146→  src: string; title: string; onSwitchServer: () => void; serverIndex: number; totalServers: number;
   147→}) {
   148→  const [state, setState] = useState<'loading' | 'ready' | 'error' | 'timeout'>('loading');
   149→  const [failedServers, setFailedServers] = useState<Set<number>>(new Set());
   150→  const [autoFallbackAttempted, setAutoFallbackAttempted] = useState(false);
   151→  const [allFailed, setAllFailed] = useState(false);
   152→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   153→
   154→  useEffect(() => {
   155→    setState('loading');
   156→    setAutoFallbackAttempted(false);
   157→    if (timerRef.current) clearTimeout(timerRef.current);
   158→    timerRef.current = setTimeout(() => {
   159→      setState(prev => prev === 'loading' ? 'timeout' : prev);
   160→      setFailedServers(prev => new Set(prev).add(serverIndex));
   161→    }, 15000);
   162→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   163→  }, [src, serverIndex]);
   164→
   165→  // Auto-fallback when server fails
   166→  useEffect(() => {
   167→    if ((state === 'error' || state === 'timeout') && !autoFallbackAttempted) {
   168→      setAutoFallbackAttempted(true);
   169→      const newFailed = new Set(failedServers);
   170→      newFailed.add(serverIndex);
   171→      if (newFailed.size >= totalServers) {
   172→        setAllFailed(true);
   173→        return;
   174→      }
   175→      const timer = setTimeout(() => onSwitchServer(), 1500);
   176→      return () => clearTimeout(timer);
   177→    }
   178→  }, [state, autoFallbackAttempted, failedServers, serverIndex, totalServers, onSwitchServer]);
   179→
   180→  const handleLoad = useCallback(() => {
   181→    if (timerRef.current) clearTimeout(timerRef.current);
   182→    setState('ready');
   183→    setFailedServers(prev => { const n = new Set(prev); n.delete(serverIndex); return n; });
   184→  }, [serverIndex]);
   185→
   186→  const handleError = useCallback(() => {
   187→    if (timerRef.current) clearTimeout(timerRef.current);
   188→    setState('error');
   189→    setFailedServers(prev => new Set(prev).add(serverIndex));
   190→  }, [serverIndex]);
   191→
   192→  const showOverlay = state === 'error' || state === 'timeout';
   193→  const serverName = Object.values(API_CONFIG.videoServers)[serverIndex]?.name || `Serveur ${serverIndex + 1}`;
   194→  const available = totalServers - failedServers.size;
   195→  const isAutoFallingBack = showOverlay && !allFailed && !autoFallbackAttempted;
   196→
   197→  return (
   198→    <div className="absolute inset-0 flex items-center justify-center">
   199→      <iframe
   200→        src={src}
   201→        className={cn('w-full h-full transition-opacity duration-500', showOverlay ? 'opacity-0' : 'opacity-100')}
   202→        style={{ aspectRatio: '16/9', minHeight: '100%', maxWidth: '177.78%' }}
   203→        allowFullScreen
   204→        allow="autoplay; fullscreen; picture-in-picture"
   205→        referrerPolicy="origin"
   206→        title={title}
   207→        onLoad={handleLoad}
   208→        onError={handleError}
   209→      />
   210→      {state === 'loading' && !showOverlay && (
   211→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-black/60">
   212→          <div className="flex flex-col items-center gap-2">
   213→            <div className="w-6 h-6 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   214→            <span className="text-[10px] text-white/30">{serverName}</span>
   215→          </div>
   216→        </div>
   217→      )}
   218→      {/* Auto-fallback transition */}
   219→      {isAutoFallingBack && (
   220→        <div className="absolute inset-0 flex items-center justify-center z-15 bg-black/80">
   221→          <div className="flex flex-col items-center gap-2 animate-pulse">
   222→            <Wifi className="w-6 h-6 text-primary" />
   223→            <p className="text-[11px] text-white/50">Serveur suivant...</p>
   224→          </div>
   225→        </div>
   226→      )}
   227→      {/* All servers failed */}
   228→      {showOverlay && allFailed && (
   229→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   230→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   231→            <WifiOff className="w-8 h-8 text-red-400" />
   232→            <p className="text-xs font-bold text-white">Contenu indisponible</p>
   233→            <p className="text-[10px] text-white/30">Aucun serveur ne peut charger ce contenu. Réessayez plus tard.</p>
   234→            <button onClick={() => { setAllFailed(false); setFailedServers(new Set()); onSwitchServer(); }}
   235→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   236→              <RefreshCw className="w-3.5 h-3.5" />
   237→              Réessayer
   238→            </button>
   239→          </div>
   240→        </div>
   241→      )}
   242→      {/* Some servers still available */}
   243→      {showOverlay && !allFailed && !isAutoFallingBack && (
   244→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   245→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   246→            <AlertTriangle className="w-8 h-8 text-amber-400" />
   247→            <p className="text-xs font-bold text-white">Serveur indisponible</p>
   248→            <p className="text-[10px] text-white/30">{serverName} ne répond pas</p>
   249→            <p className="text-[10px] text-primary/50">{available} serveur{available > 1 ? 's' : ''} restant{available > 1 ? 's' : ''}</p>
   250→            <button onClick={onSwitchServer}
   251→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   252→              <SkipForward className="w-3.5 h-3.5" />
   253→              Serveur suivant
   254→            </button>
   255→            <p className="text-[9px] text-white/15">{serverIndex + 1} / {totalServers} · {failedServers.size} échoué{failedServers.size > 1 ? 's' : ''}</p>
   256→          </div>
   257→        </div>
   258→      )}
   259→    </div>
   260→  );
   261→}

   273→// ─── Load shorts from a single endpoint, deduped ───
   274→async function loadShortsFromEndpoint(
   275→  endpoint: string,
   276→  mediaType: 'movie' | 'tv',
   277→  genreLabel?: string,
   278→  maxItems: number = 8,
   279→  seenIds: Set<string> = new Set()
   280→): Promise<ShortItem[]> {
   281→  const data = await fetchTMDB<TMDBResponse<Media>>(endpoint);
   282→  if (!data?.results) return [];
   283→
   284→  const items: ShortItem[] = [];
   285→
   286→  for (const item of data.results) {
   287→    if (items.length >= maxItems) break;
   288→
   289→    // Dedup key: same id + same mediaType = skip
   290→    const dedupKey = `${item.id}-${mediaType}`;
   291→    if (seenIds.has(dedupKey)) continue;
   292→    seenIds.add(dedupKey);
   293→
   294→    if (mediaType === 'tv') {
   295→      // For TV: fetch first episode
   296→      const detail = await fetchTMDB<any>(`/tv/${item.id}`);
   297→      if (detail?.seasons) {
   298→        const firstRealSeason = detail.seasons.find((s: any) => s.season_number > 0);
   299→        if (firstRealSeason) {
   300→          const seasonData = await fetchTMDB<any>(`/tv/${item.id}/season/${firstRealSeason.season_number}`);
   301→          if (seasonData?.episodes?.[0]) {
   302→            const ep = seasonData.episodes[0];
   303→            items.push({
   304→              id: item.id,
   305→              mediaType: 'tv',
   306→              title: item.name || item.original_name || '',
   307→              overview: ep.overview || item.overview || '',
   308→              posterPath: item.poster_path,
   309→              backdropPath: ep.still_path || item.backdrop_path,
   310→              voteAverage: item.vote_average,
   311→              genreLabel,
   312→              season: firstRealSeason.season_number,
   313→              episode: ep.episode_number,
   314→              episodeName: ep.name,
   315→            });
   316→          }
   317→        }
   318→      }
   319→    } else {
   320→      // For movies: single card (no segment splitting for cleaner feed)
   321→      items.push({
   322→        id: item.id,
   323→        mediaType: 'movie',
   324→        title: item.title || item.original_title || '',
   325→        overview: item.overview || '',
   326→        posterPath: item.poster_path,
   327→        backdropPath: item.backdrop_path,
   328→        voteAverage: item.vote_average,
   329→        genreLabel,
   330→      });
   331→    }
   332→  }
   333→
   334→  return items;
   335→}

   337→// ─── Like Button with TikTok-style animation ───
   338→function LikeButton({ mediaId, mediaType, segment, season, likeTriggerRef }: { mediaId: number; mediaType: string; segment: number; season?: number | null; likeTriggerRef?: React.MutableRefObject<(() => void) | null> }) {
   339→  const [data, setData] = useState<LikeData>({ count: 0, liked: false });
   340→  const [animating, setAnimating] = useState(false);
   341→  const [justLiked, setJustLiked] = useState(false); // track if we JUST liked for animation direction
   342→  const [isToggling, setIsToggling] = useState(false); // prevent double-clicks
   343→  const { data: session } = useSession();
   344→  const normalizedSeason = season || 0;
   345→
   346→  useEffect(() => {
   347→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   348→    fetch(`/api/shorts/like?${params}`)
   349→      .then(r => r.ok ? r.json() : null)
   350→      .then(d => { if (d?.count !== undefined) setData(d); })
   351→      .catch(() => {});
   352→  }, [mediaId, mediaType, segment, normalizedSeason]);
   353→
   354→  const toggleLike = async () => {
   355→    if (!session || isToggling) return;
   356→    setIsToggling(true);
   357→    
   358→    // Optimistic update — immediately show the animation
   359→    const willLike = !data.liked;
   360→    setJustLiked(willLike);
   361→    setAnimating(true);
   362→    setData(prev => ({ count: prev.count + (willLike ? 1 : -1), liked: willLike }));
   363→
   364→    try {
   365→      const res = await fetch('/api/shorts/like', {
   366→        method: 'POST',
   367→        headers: { 'Content-Type': 'application/json' },
   368→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason }),
   369→      });
   370→      
   371→      if (!res.ok) {
   372→        // HTTP error (401, 500, etc.) — revert optimistic update
   373→        console.warn('[like] HTTP error:', res.status);
   374→        setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   375→      } else {
   376→        const d = await res.json();
   377→        if (d?.liked !== undefined) {
   378→          // Server confirmed — use server state
   379→          setData(prev => ({ count: d.count ?? prev.count, liked: d.liked }));
   380→        } else if (d?.error) {
   381→          // Server error — revert optimistic update
   382→          console.warn('[like] API error:', d.error);
   383→          setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   384→        }
   385→      }
   386→    } catch (err) {
   387→      // Network error — revert optimistic update
   388→      console.warn('[like] Network error:', err);
   389→      setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   390→    }
   391→    setTimeout(() => { setAnimating(false); setJustLiked(false); setIsToggling(false); }, 600);
   392→  };
   393→
   394→  // Expose toggleLike via ref so parent (double-tap) can trigger it
   395→  useEffect(() => {
   396→    if (likeTriggerRef) likeTriggerRef.current = toggleLike;
   397→  }); // update every render to keep ref fresh
   398→
   399→  return (
   400→    <button onClick={toggleLike} className="flex flex-col items-center gap-0.5 group active:scale-90 transition-transform">
   401→      <div className="relative">
   402→        <Heart
   403→          className={cn(
   404→            'w-7 h-7 transition-all duration-300',
   405→            animating && justLiked && 'scale-125',
   406→            data.liked
   407→              ? 'text-red-500 fill-red-500 drop-shadow-[0_0_8px_rgba(239,68,68,0.4)]'
   408→              : 'text-white/80 group-hover:text-white'
   409→          )}
   410→          strokeWidth={data.liked ? 0 : 1.5}
   411→        />
   412→        {animating && (
   413→          <div className="absolute inset-0 flex items-center justify-center">
   414→            <div className={cn(
   415→              'w-12 h-12 rounded-full animate-ping-once',
   416→              justLiked ? 'bg-red-500/25' : 'bg-white/10'
   417→            )} />
   418→          </div>
   419→        )}
   420→      </div>
   421→      <span className={cn('text-[11px] font-medium transition-colors', data.liked ? 'text-red-400' : 'text-white/40')}>
   422→        {data.count > 0 ? data.count : 'J\'aime'}
   423→      </span>
   424→    </button>
   425→  );
   426→}

   429→function CommentSheet({ mediaId, mediaType, segment, season, onClose }: {
   430→  mediaId: number; mediaType: string; segment: number; season?: number | null; onClose: () => void;
   431→}) {
   432→  const [comments, setComments] = useState<CommentData[]>([]);
   433→  const [newComment, setNewComment] = useState('');
   434→  const [submitting, setSubmitting] = useState(false);
   435→  const [visible, setVisible] = useState(false);
   436→  const { data: session } = useSession();
   437→  const normalizedSeason = season || 0;
   438→
   439→  useEffect(() => { requestAnimationFrame(() => setVisible(true)); }, []);
   440→
   441→  useEffect(() => {
   442→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   443→    fetch(`/api/shorts/comments?${params}`)
   444→      .then(r => r?.json())
   445→      .then(d => { if (d?.comments) setComments(d.comments); })
   446→      .catch(() => {});
   447→  }, [mediaId, mediaType, segment, normalizedSeason]);
   448→
   449→  const handleClose = () => { setVisible(false); setTimeout(onClose, 250); };
   450→  const handleBackdropClick = (e: React.MouseEvent) => { if (e.target === e.currentTarget) handleClose(); };
   451→
   452→  const submitComment = async () => {
   453→    if (!session || !newComment.trim() || submitting) return;
   454→    setSubmitting(true);
   455→    try {
   456→      const res = await fetch('/api/shorts/comments', {
   457→        method: 'POST',
   458→        headers: { 'Content-Type': 'application/json' },
   459→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason, content: newComment.trim() }),
   460→      });
   461→      const d = await res.json();
   462→      if (d?.comment) { setComments(prev => [d.comment, ...prev]); setNewComment(''); }
   463→    } catch {}
   464→    setSubmitting(false);
   465→  };
   466→
   467→  return (
   468→    <div className="absolute inset-0 z-50 flex flex-col justify-end" onClick={handleBackdropClick}>
   469→      <div className={cn('absolute inset-0 bg-black/40 transition-opacity duration-250', visible ? 'opacity-100' : 'opacity-0')} />
   470→      <div className={cn(
   471→        'relative z-10 flex flex-col rounded-t-2xl transition-transform duration-250 ease-out',
   472→        'bg-[#1a1a1a]/95 backdrop-blur-xl border-t border-white/[0.06]',
   473→        visible ? 'translate-y-0' : 'translate-y-full'
   474→      )} style={{ height: '55vh', maxHeight: '55dvh' }} onClick={e => e.stopPropagation()}>
   475→        <div className="flex justify-center pt-2.5 pb-1"><div className="w-8 h-1 rounded-full bg-white/15" /></div>
   476→        <div className="flex items-center justify-between px-4 pb-2.5 border-b border-white/[0.06]">
   477→          <h3 className="text-[13px] font-bold text-white/90">Commentaires<span className="ml-1.5 text-white/30 font-normal">{comments.length}</span></h3>
   478→          <button onClick={handleClose} className="p-1.5 rounded-full hover:bg-white/5 transition-colors"><X className="w-4 h-4 text-white/40" /></button>
   479→        </div>
   480→        <div className="flex-1 overflow-y-auto px-4 py-3 space-y-3 scrollbar-hide">

   913→          const animeShows = animeTv.results.slice(0, 5).filter((show: Media) => {
   914→            const key = `${show.id}-tv`;
   915→            if (seen.has(key)) return false;
   916→            seen.add(key);
   917→            return true;
   918→          });
   919→          const animeResults = await Promise.allSettled(animeShows.map((show: Media) => fetchTvEpisode(show, seen)));
   920→          for (const r of animeResults) {
   921→            if (r.status === 'fulfilled' && r.value) {
   922→              r.value.genreLabel = 'Anime';
   923→              items.push(r.value);
   924→            }
   925→          }
   926→        }
   927→
   928→        // Shuffle for variety
   929→        items.sort(() => Math.random() - 0.5);
   930→
   931→      } else if (tab === 'series') {
   932→        // Series tab — profile-filtered TV shows (parallel fetching)
   933→        const seriesEndpoint = profileGenreParams
   934→          ? `/discover/tv?sort_by=popularity.desc&${profileGenreParams}`
   935→          : `/tv/popular`;
   936→        const data = await fetchTMDB<TMDBResponse<Media>>(withPage(seriesEndpoint, page));
   937→        if (data?.results) {
   938→          const shows = data.results.filter((show: Media) => {
   939→            const key = `${show.id}-tv`;
   940→            if (seen.has(key)) return false;
   941→            seen.add(key);
   942→            return true;
   943→          });
   944→          const results = await Promise.allSettled(shows.map((show: Media) => fetchTvEpisode(show, seen)));
   945→          for (const r of results) {
   946→            if (r.status === 'fulfilled' && r.value) items.push(r.value);
   947→          }
   948→        }
   949→
   950→      } else if (tab === 'films') {
   951→        // Films tab — profile-filtered movies
   952→        const filmsEndpoint = profileGenreParams
   953→          ? `/discover/movie?sort_by=popularity.desc&${profileGenreParams}`
   954→          : `/movie/popular`;
   955→        const data = await fetchTMDB<TMDBResponse<Media>>(withPage(filmsEndpoint, page));
   956→        if (data?.results) {
   957→          for (const movie of data.results) {
   958→            const key = `${movie.id}-movie`;
   959→            if (seen.has(key)) continue;
   960→            seen.add(key);
   961→            items.push({
   962→              id: movie.id, mediaType: 'movie',
   963→              title: movie.title || movie.original_title || '',
   964→              overview: movie.overview || '', posterPath: movie.poster_path,
   965→              backdropPath: movie.backdrop_path, voteAverage: movie.vote_average,
   966→            });
   967→          }
   968→        }
   969→
   970→      } else if (tab === 'anime') {
   971→        // Anime: both TV and movies with animation genre + anime keyword
   972→        const [animeTv, animeMovies] = await Promise.all([
   973→          fetchTMDB<TMDBResponse<Media>>(withPage(`/discover/tv?with_genres=16&with_keywords=210024&sort_by=popularity.desc`, page)),
   974→          fetchTMDB<TMDBResponse<Media>>(withPage(`/discover/movie?with_genres=16&with_keywords=210024&sort_by=popularity.desc`, page)),
   975→        ]);
   976→
   977→        if (animeTv?.results) {
   978→          const shows = animeTv.results.filter((show: Media) => {
   979→            const key = `${show.id}-tv`;
   980→            if (seen.has(key)) return false;
   981→            seen.add(key);
   982→            return true;
   983→          });
   984→          const results = await Promise.allSettled(shows.map((show: Media) => fetchTvEpisode(show, seen)));
   985→          for (const r of results) {
   986→            if (r.status === 'fulfilled' && r.value) {
   987→              r.value.genreLabel = 'Anime';
   988→              items.push(r.value);
   989→            }
   990→          }
   991→        }
   992→
   993→        if (animeMovies?.results) {
   994→          for (const movie of animeMovies.results) {
   995→            const key = `${movie.id}-movie`;
   996→            if (seen.has(key)) continue;
   997→            seen.add(key);
   998→            items.push({
   999→              id: movie.id, mediaType: 'movie',
  1000→              title: movie.title || movie.original_title || '',
  1001→              overview: movie.overview || '', posterPath: movie.poster_path,
  1002→              backdropPath: movie.backdrop_path, voteAverage: movie.vote_average,
  1003→              genreLabel: 'Anime',
  1004→            });
  1005→          }
  1006→        }
  1007→
  1008→        items.sort(() => Math.random() - 0.5);
  1009→      }
  1010→    } catch (error) {
  1011→      console.error('[shorts] Load error:', error);
  1012→    }
  1013→
  1014→    if (append) {
  1015→      setShorts(prev => [...prev, ...items]);
  1016→    } else {
  1017→      setShorts(items);
  1018→      setActiveIndex(0);
  1019→    }
  1020→
  1021→    setLoading(false);
  1022→    setLoadingMore(false);
  1023→    loadingMoreRef.current = false;
  1024→  }, [profile?.type, getDiscoverEndpoint]);
  1025→
  1026→  // Load on auth/tab change — supports both authenticated and guest users
  1027→  useEffect(() => {
  1028→    const isReady = status === 'authenticated' || (status === 'unauthenticated' && isGuest);
  1029→    if (isReady) {
  1030→      pageRef.current = 1;
  1031→      seenIdsRef.current = new Set();
  1032→      loadShorts(activeTab, 1, false);
  1033→    }
  1034→  }, [status, isGuest, profile, activeTab]); // eslint-disable-line react-hooks/exhaustive-deps
  1035→
  1036→  // Infinite scroll — load more when near bottom
  1037→  useEffect(() => {
  1038→    const el = scrollRef.current;
  1039→    if (!el) return;
  1040→
  1041→    const handleScroll = () => {
  1042→      const { scrollTop, scrollHeight, clientHeight } = el;
  1043→      if (scrollHeight - scrollTop - clientHeight < 800 && !loadingMoreRef.current) {
  1044→        pageRef.current += 1;
  1045→        loadShorts(activeTab, pageRef.current, true);
  1046→      }
  1047→    };
  1048→
  1049→    el.addEventListener('scroll', handleScroll, { passive: true });
  1050→    return () => el.removeEventListener('scroll', handleScroll);
  1051→  }, [activeTab, loadShorts]);
  1052→
  1053→  // Track active card on scroll
  1054→  useEffect(() => {
  1055→    const el = scrollRef.current;
  1056→    if (!el) return;
  1057→    const observer = new IntersectionObserver(
  1058→      (entries) => {
  1059→        entries.forEach(entry => {
  1060→          if (entry.isIntersecting) {
  1061→            const idx = Number(entry.target.getAttribute('data-index'));
  1062→            if (!isNaN(idx)) setActiveIndex(idx);
  1063→          }
  1064→        });
  1065→      },
  1066→      { root: el, threshold: 0.6 }
  1067→    );
  1068→    el.querySelectorAll('[data-index]').forEach(card => observer.observe(card));
  1069→    return () => observer.disconnect();
  1070→  }, [shorts]);
  1071→
  1072→  // Handle search result click — insert at current position
  1073→  const handleSearchResult = (item: ShortItem) => {
  1074→    setShorts(prev => {
  1075→      const newShorts = [...prev];
  1076→      newShorts.splice(activeIndex + 1, 0, item);
  1077→      return newShorts;
  1078→    });
  1079→  };
  1080→
  1081→  if (status === 'loading') {
  1082→    return (
  1083→      <div className="h-screen bg-background flex items-center justify-center">
  1084→        <div className="w-10 h-10 border-3 border-primary border-t-transparent rounded-full animate-spin" />
  1085→      </div>
  1086→    );
  1087→  }
  1088→
  1089→  return (
  1090→    <div className="h-screen bg-black relative overflow-hidden">
  1091→      {/* Top Bar */}
  1092→      <div className="absolute top-0 left-0 right-0 z-40 flex items-center justify-between px-4 pt-[env(safe-area-inset-top,12px)] pb-2">
  1093→        <button onClick={() => router.push('/')} className="p-2 rounded-full bg-black/30 backdrop-blur-sm hover:bg-black/50 transition-colors">
  1094→          <ArrowLeft className="w-5 h-5 text-white/80" />
  1095→        </button>
  1096→
  1097→        {/* Tab selector */}
  1098→        <div className="flex items-center gap-0.5 bg-black/30 backdrop-blur-sm rounded-full p-0.5">
  1099→          {CONTENT_TABS.map(tab => (
  1100→            <button
  1101→              key={tab.key}
  1102→              onClick={() => setActiveTab(tab.key)}
  1103→              className={cn(
  1104→                'px-2.5 py-1 rounded-full text-[10px] sm:text-[11px] font-semibold transition-all whitespace-nowrap',
  1105→                activeTab === tab.key
  1106→                  ? 'bg-primary text-black'
  1107→                  : 'text-white/50 hover:text-white/70'
  1108→              )}
  1109→            >
  1110→              {tab.label}
  1111→            </button>
  1112→          ))}
  1113→        </div>
  1114→
  1115→        {/* Search toggle */}
  1116→        <button
  1117→          onClick={() => setShowSearch(!showSearch)}
  1118→          className={cn(
  1119→            'p-2 rounded-full backdrop-blur-sm transition-colors',
  1120→            showSearch ? 'bg-primary/30 text-primary' : 'bg-black/30 text-white/70 hover:bg-black/50'
  1121→          )}
  1122→        >
  1123→          <Search className="w-5 h-5" />
  1124→        </button>
  1125→      </div>
  1126→
  1127→      {/* Shorts Feed — Vertical Snap Scroll with Infinite Loading */}
  1128→      {loading ? (
  1129→        <div className="h-full flex items-center justify-center">
  1130→          <div className="flex flex-col items-center gap-3">
  1131→            <div className="w-8 h-8 border-2 border-primary border-t-transparent rounded-full animate-spin" />
  1132→            <span className="text-xs text-white/30">Chargement des shorts...</span>
  1133→          </div>
  1134→        </div>
  1135→      ) : shorts.length === 0 ? (
  1136→        <div className="h-full flex items-center justify-center">
  1137→          <div className="flex flex-col items-center gap-3">
  1138→            <Zap className="w-10 h-10 text-white/10" />
  1139→            <p className="text-sm text-white/30">Aucun short disponible</p>
  1140→            <p className="text-[11px] text-white/20">Essayez un autre onglet</p>
  1141→          </div>
  1142→        </div>
  1143→      ) : (
  1144→        <div
  1145→          ref={scrollRef}
  1146→          className="h-full overflow-y-scroll snap-y snap-mandatory scrollbar-hide"
  1147→          style={{ WebkitOverflowScrolling: 'touch' }}
  1148→        >
  1149→          {shorts.map((item, i) => (
  1150→            <div key={`${item.id}-${item.mediaType}-${item.segmentIndex ?? ''}-${item.season ?? ''}-${item.episode ?? ''}-${i}`} data-index={i} className="h-full">
  1151→              <ShortCard item={item} isActive={i === activeIndex} />
  1152→            </div>
  1153→          ))}
  1154→          {/* Loading more indicator */}
  1155→          {loadingMore && (
  1156→            <div className="h-20 flex items-center justify-center">
  1157→              <div className="w-6 h-6 border-2 border-primary border-t-transparent rounded-full animate-spin" />
  1158→            </div>
  1159→          )}
  1160→        </div>
  1161→      )}
  1162→
  1163→      {/* Search overlay */}
  1164→      {showSearch && (
  1165→        <ShortsSearch onClose={() => setShowSearch(false)} onResultClick={handleSearchResult} />
  1166→      )}
  1167→    </div>
  1168→  );
  1169→}
