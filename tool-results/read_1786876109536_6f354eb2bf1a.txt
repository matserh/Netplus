    61→    61→function formatTimeShort(seconds: number): string {
    62→    62→  const m = Math.floor(seconds / 60);
    63→    63→  const s = seconds % 60;
    64→    64→  return `${m}:${s.toString().padStart(2, '0')}`;
    65→    65→}

    67→    67→function formatRelativeDate(dateStr: string): string {
    68→    68→  const diff = Date.now() - new Date(dateStr).getTime();
    69→    69→  const min = Math.floor(diff / 60000);
    70→    70→  if (min < 1) return "À l'instant";
    71→    71→  if (min < 60) return `Il y a ${min}min`;
    72→    72→  const h = Math.floor(min / 60);
    73→    73→  if (h < 24) return `Il y a ${h}h`;
    74→    74→  const d = Math.floor(h / 24);
    75→    75→  if (d < 7) return `Il y a ${d}j`;
    76→    76→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' });
    77→    77→}

    79→    79→// ─── Segment generator for movies ───
    80→    80→function generateMovieSegments(runtime: number): { index: number; title: string; start: number }[] {
    81→    81→  if (!runtime || runtime <= 0) return [{ index: 0, title: 'Début', start: 0 }];
    82→    82→  const segmentLength = 5;
    83→    83→  const segments = [];
    84→    84→  for (let i = 0; i < Math.ceil(runtime / segmentLength); i++) {
    85→    85→    const startMin = i * segmentLength;
    86→    86→    const endMin = Math.min((i + 1) * segmentLength, runtime);
    87→    87→    segments.push({
    88→    88→      index: i,
    89→    89→      title: `${formatTimeShort(startMin * 60)} - ${formatTimeShort(endMin * 60)}`,
    90→    90→      start: startMin * 60,
    91→    91→    });
    92→    92→  }
    93→    93→  return segments;
    94→    94→}

   108→   108→function SmartShortPlayer({ src, title, onSwitchServer, serverIndex, totalServers }: {
   109→   109→  src: string; title: string; onSwitchServer: () => void; serverIndex: number; totalServers: number;
   110→   110→}) {
   111→   111→  const [state, setState] = useState<'loading' | 'ready' | 'error' | 'timeout'>('loading');
   112→   112→  const [failedServers, setFailedServers] = useState<Set<number>>(new Set());
   113→   113→  const [autoFallbackAttempted, setAutoFallbackAttempted] = useState(false);
   114→   114→  const [allFailed, setAllFailed] = useState(false);
   115→   115→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   116→   116→
   117→   117→  useEffect(() => {
   118→   118→    setState('loading');
   119→   119→    setAutoFallbackAttempted(false);
   120→   120→    if (timerRef.current) clearTimeout(timerRef.current);
   121→   121→    timerRef.current = setTimeout(() => {
   122→   122→      setState(prev => prev === 'loading' ? 'timeout' : prev);
   123→   123→      setFailedServers(prev => new Set(prev).add(serverIndex));
   124→   124→    }, 15000);
   125→   125→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   126→   126→  }, [src, serverIndex]);
   127→   127→
   128→   128→  // Auto-fallback when server fails
   129→   129→  useEffect(() => {
   130→   130→    if ((state === 'error' || state === 'timeout') && !autoFallbackAttempted) {
   131→   131→      setAutoFallbackAttempted(true);
   132→   132→      const newFailed = new Set(failedServers);
   133→   133→      newFailed.add(serverIndex);
   134→   134→      if (newFailed.size >= totalServers) {
   135→   135→        setAllFailed(true);
   136→   136→        return;
   137→   137→      }
   138→   138→      const timer = setTimeout(() => onSwitchServer(), 1500);
   139→   139→      return () => clearTimeout(timer);
   140→   140→    }
   141→   141→  }, [state, autoFallbackAttempted, failedServers, serverIndex, totalServers, onSwitchServer]);
   142→   142→
   143→   143→  const handleLoad = useCallback(() => {
   144→   144→    if (timerRef.current) clearTimeout(timerRef.current);
   145→   145→    setState('ready');
   146→   146→    setFailedServers(prev => { const n = new Set(prev); n.delete(serverIndex); return n; });
   147→   147→  }, [serverIndex]);
   148→   148→
   149→   149→  const handleError = useCallback(() => {
   150→   150→    if (timerRef.current) clearTimeout(timerRef.current);
   151→   151→    setState('error');
   152→   152→    setFailedServers(prev => new Set(prev).add(serverIndex));
   153→   153→  }, [serverIndex]);
   154→   154→
   155→   155→  const showOverlay = state === 'error' || state === 'timeout';
   156→   156→  const serverName = Object.values(API_CONFIG.videoServers)[serverIndex]?.name || `Serveur ${serverIndex + 1}`;
   157→   157→  const available = totalServers - failedServers.size;
   158→   158→  const isAutoFallingBack = showOverlay && !allFailed && !autoFallbackAttempted;
   159→   159→
   160→   160→  return (
   161→   161→    <div className="absolute inset-0 flex items-center justify-center">
   162→   162→      <iframe
   163→   163→        src={src}
   164→   164→        className={cn('w-full h-full transition-opacity duration-500', showOverlay ? 'opacity-0' : 'opacity-100')}
   165→   165→        style={{ aspectRatio: '16/9', minHeight: '100%', maxWidth: '177.78%' }}
   166→   166→        allowFullScreen
   167→   167→        allow="autoplay; fullscreen; picture-in-picture"
   168→   168→        referrerPolicy="origin"
   169→   169→        title={title}
   170→   170→        onLoad={handleLoad}
   171→   171→        onError={handleError}
   172→   172→      />
   173→   173→      {state === 'loading' && !showOverlay && (
   174→   174→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-black/60">
   175→   175→          <div className="flex flex-col items-center gap-2">
   176→   176→            <div className="w-6 h-6 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   177→   177→            <span className="text-[10px] text-white/30">{serverName}</span>
   178→   178→          </div>
   179→   179→        </div>
   180→   180→      )}
   181→   181→      {/* Auto-fallback transition */}
   182→   182→      {isAutoFallingBack && (
   183→   183→        <div className="absolute inset-0 flex items-center justify-center z-15 bg-black/80">
   184→   184→          <div className="flex flex-col items-center gap-2 animate-pulse">
   185→   185→            <Wifi className="w-6 h-6 text-primary" />
   186→   186→            <p className="text-[11px] text-white/50">Serveur suivant...</p>
   187→   187→          </div>
   188→   188→        </div>
   189→   189→      )}
   190→   190→      {/* All servers failed */}
   191→   191→      {showOverlay && allFailed && (
   192→   192→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   193→   193→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   194→   194→            <WifiOff className="w-8 h-8 text-red-400" />
   195→   195→            <p className="text-xs font-bold text-white">Contenu indisponible</p>
   196→   196→            <p className="text-[10px] text-white/30">Aucun serveur ne peut charger ce contenu. Réessayez plus tard.</p>
   197→   197→            <button onClick={() => { setAllFailed(false); setFailedServers(new Set()); onSwitchServer(); }}
   198→   198→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   199→   199→              <RefreshCw className="w-3.5 h-3.5" />
   200→   200→              Réessayer
   201→   201→            </button>
   202→   202→          </div>
   203→   203→        </div>
   204→   204→      )}
   205→   205→      {/* Some servers still available */}
   206→   206→      {showOverlay && !allFailed && !isAutoFallingBack && (
   207→   207→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   208→   208→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   209→   209→            <AlertTriangle className="w-8 h-8 text-amber-400" />
   210→   210→            <p className="text-xs font-bold text-white">Serveur indisponible</p>
   211→   211→            <p className="text-[10px] text-white/30">{serverName} ne répond pas</p>
   212→   212→            <p className="text-[10px] text-primary/50">{available} serveur{available > 1 ? 's' : ''} restant{available > 1 ? 's' : ''}</p>
   213→   213→            <button onClick={onSwitchServer}
   214→   214→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   215→   215→              <SkipForward className="w-3.5 h-3.5" />
   216→   216→              Serveur suivant
   217→   217→            </button>
   218→   218→            <p className="text-[9px] text-white/15">{serverIndex + 1} / {totalServers} · {failedServers.size} échoué{failedServers.size > 1 ? 's' : ''}</p>
   219→   219→          </div>
   220→   220→        </div>
   221→   221→      )}
   222→   222→    </div>
   223→   223→  );
   224→   224→}

   236→   236→// ─── Load shorts from a single endpoint, deduped ───
   237→   237→async function loadShortsFromEndpoint(
   238→   238→  endpoint: string,
   239→   239→  mediaType: 'movie' | 'tv',
   240→   240→  genreLabel?: string,
   241→   241→  maxItems: number = 8,
   242→   242→  seenIds: Set<string> = new Set()
   243→   243→): Promise<ShortItem[]> {
   244→   244→  const data = await fetchTMDB<TMDBResponse<Media>>(endpoint);
   245→   245→  if (!data?.results) return [];
   246→   246→
   247→   247→  const items: ShortItem[] = [];
   248→   248→
   249→   249→  for (const item of data.results) {
   250→   250→    if (items.length >= maxItems) break;
   251→   251→
   252→   252→    // Dedup key: same id + same mediaType = skip
   253→   253→    const dedupKey = `${item.id}-${mediaType}`;
   254→   254→    if (seenIds.has(dedupKey)) continue;
   255→   255→    seenIds.add(dedupKey);
   256→   256→
   257→   257→    if (mediaType === 'tv') {
   258→   258→      // For TV: fetch first episode
   259→   259→      const detail = await fetchTMDB<any>(`/tv/${item.id}`);
   260→   260→      if (detail?.seasons) {
   261→   261→        const firstRealSeason = detail.seasons.find((s: any) => s.season_number > 0);
   262→   262→        if (firstRealSeason) {
   263→   263→          const seasonData = await fetchTMDB<any>(`/tv/${item.id}/season/${firstRealSeason.season_number}`);
   264→   264→          if (seasonData?.episodes?.[0]) {
   265→   265→            const ep = seasonData.episodes[0];
   266→   266→            items.push({
   267→   267→              id: item.id,
   268→   268→              mediaType: 'tv',
   269→   269→              title: item.name || item.original_name || '',
   270→   270→              overview: ep.overview || item.overview || '',
   271→   271→              posterPath: item.poster_path,
   272→   272→              backdropPath: ep.still_path || item.backdrop_path,
   273→   273→              voteAverage: item.vote_average,
   274→   274→              genreLabel,
   275→   275→              season: firstRealSeason.season_number,
   276→   276→              episode: ep.episode_number,
   277→   277→              episodeName: ep.name,
   278→   278→            });
   279→   279→          }
   280→   280→        }
   281→   281→      }
   282→   282→    } else {
   283→   283→      // For movies: single card (no segment splitting for cleaner feed)
   284→   284→      items.push({
   285→   285→        id: item.id,
   286→   286→        mediaType: 'movie',
   287→   287→        title: item.title || item.original_title || '',
   288→   288→        overview: item.overview || '',
   289→   289→        posterPath: item.poster_path,
   290→   290→        backdropPath: item.backdrop_path,
   291→   291→        voteAverage: item.vote_average,
   292→   292→        genreLabel,
   293→   293→      });
   294→   294→    }
   295→   295→  }
   296→   296→
   297→   297→  return items;
   298→   298→}

   301→   301→function LikeButton({ mediaId, mediaType, segment, season, likeTriggerRef }: { mediaId: number; mediaType: string; segment: number; season?: number | null; likeTriggerRef?: React.MutableRefObject<(() => void) | null> }) {
   302→   302→  const [data, setData] = useState<LikeData>({ count: 0, liked: false });
   303→   303→  const [animating, setAnimating] = useState(false);
   304→   304→  const [justLiked, setJustLiked] = useState(false); // track if we JUST liked for animation direction
   305→   305→  const [isToggling, setIsToggling] = useState(false); // prevent double-clicks
   306→   306→  const { data: session } = useSession();
   307→   307→  const normalizedSeason = season || 0;
   308→   308→
   309→   309→  useEffect(() => {
   310→   310→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   311→   311→    apiFetch(`/api/shorts/like?${params}`)
   312→   312→      .then(r => r.ok ? r.json() : null)
   313→   313→      .then(d => { if (d?.count !== undefined) setData(d); })
   314→   314→      .catch(() => {});
   315→   315→  }, [mediaId, mediaType, segment, normalizedSeason]);
   316→   316→
   317→   317→  const toggleLike = async () => {
   318→   318→    if (!session || isToggling) return;
   319→   319→    setIsToggling(true);
   320→   320→    
   321→   321→    // Optimistic update — immediately show the animation
   322→   322→    const willLike = !data.liked;
   323→   323→    setJustLiked(willLike);
   324→   324→    setAnimating(true);
   325→   325→    setData(prev => ({ count: prev.count + (willLike ? 1 : -1), liked: willLike }));
   326→   326→
   327→   327→    try {
   328→   328→      const res = await apiFetch('/api/shorts/like', {
   329→   329→        method: 'POST',
   330→   330→        headers: { 'Content-Type': 'application/json' },
   331→   331→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason }),
   332→   332→      });
   333→   333→      
   334→   334→      if (!res.ok) {
   335→   335→        // HTTP error (401, 500, etc.) — revert optimistic update
   336→   336→        console.warn('[like] HTTP error:', res.status);
   337→   337→        setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   338→   338→      } else {
   339→   339→        const d = await res.json();
   340→   340→        if (d?.liked !== undefined) {
   341→   341→          // Server confirmed — use server state
   342→   342→          setData(prev => ({ count: d.count ?? prev.count, liked: d.liked }));
   343→   343→        } else if (d?.error) {
   344→   344→          // Server error — revert optimistic update
   345→   345→          console.warn('[like] API error:', d.error);
   346→   346→          setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   347→   347→        }
   348→   348→      }
   349→   349→    } catch (err) {
   350→   350→      // Network error — revert optimistic update
   351→   351→      console.warn('[like] Network error:', err);
   352→   352→      setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   353→   353→    }
   354→   354→    setTimeout(() => { setAnimating(false); setJustLiked(false); setIsToggling(false); }, 600);
   355→   355→  };
   356→   356→
   357→   357→  // Expose toggleLike via ref so parent (double-tap) can trigger it
   358→   358→  useEffect(() => {
   359→   359→    if (likeTriggerRef) likeTriggerRef.current = toggleLike;
   360→   360→  }); // update every render to keep ref fresh
   361→   361→
   362→   362→  return (
   363→   363→    <button onClick={toggleLike} className="flex flex-col items-center gap-0.5 group active:scale-90 transition-transform">
   364→   364→      <div className="relative">
   365→   365→        <Heart
   366→   366→          className={cn(
   367→   367→            'w-7 h-7 transition-all duration-300',
   368→   368→            animating && justLiked && 'scale-125',
   369→   369→            data.liked
   370→   370→              ? 'text-red-500 fill-red-500 drop-shadow-[0_0_8px_rgba(239,68,68,0.4)]'
   371→   371→              : 'text-white/80 group-hover:text-white'
   372→   372→          )}
   373→   373→          strokeWidth={data.liked ? 0 : 1.5}
   374→   374→        />
   375→   375→        {animating && (
   376→   376→          <div className="absolute inset-0 flex items-center justify-center">
   377→   377→            <div className={cn(
   378→   378→              'w-12 h-12 rounded-full animate-ping-once',
   379→   379→              justLiked ? 'bg-red-500/25' : 'bg-white/10'
   380→   380→            )} />
   381→   381→          </div>
   382→   382→        )}
   383→   383→      </div>
   384→   384→      <span className={cn('text-[11px] font-medium transition-colors', data.liked ? 'text-red-400' : 'text-white/40')}>
   385→   385→        {data.count > 0 ? data.count : 'J\'aime'}
   386→   386→      </span>
   387→   387→    </button>
   388→   388→  );
   389→   389→}

   392→   392→function CommentSheet({ mediaId, mediaType, segment, season, onClose }: {
   393→   393→  mediaId: number; mediaType: string; segment: number; season?: number | null; onClose: () => void;
   394→   394→}) {
   395→   395→  const [comments, setComments] = useState<CommentData[]>([]);
   396→   396→  const [newComment, setNewComment] = useState('');
   397→   397→  const [submitting, setSubmitting] = useState(false);
   398→   398→  const [visible, setVisible] = useState(false);
   399→   399→  const { data: session } = useSession();
   400→   400→  const normalizedSeason = season || 0;
   401→   401→
   402→   402→  useEffect(() => { requestAnimationFrame(() => setVisible(true)); }, []);
   403→   403→
   404→   404→  useEffect(() => {
   405→   405→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   406→   406→    apiFetch(`/api/shorts/comments?${params}`)
   407→   407→      .then(r => r?.json())
   408→   408→      .then(d => { if (d?.comments) setComments(d.comments); })
   409→   409→      .catch(() => {});
   410→   410→  }, [mediaId, mediaType, segment, normalizedSeason]);
   411→   411→
   412→   412→  const handleClose = () => { setVisible(false); setTimeout(onClose, 250); };
   413→   413→  const handleBackdropClick = (e: React.MouseEvent) => { if (e.target === e.currentTarget) handleClose(); };
   414→   414→
   415→   415→  const submitComment = async () => {
   416→   416→    if (!session || !newComment.trim() || submitting) return;
   417→   417→    setSubmitting(true);

   777→   777→function getProfileGenreParams(profileType: string | undefined): string {

   789→   789→// ─── Helper: add page param to endpoint ───
   790→   790→function withPage(endpoint: string, page: number): string {
