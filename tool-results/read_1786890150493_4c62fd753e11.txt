    23→

    42→function WatchLoading() {
    43→  return (
    44→    <div className="min-h-screen bg-background flex items-center justify-center">
    45→      <div className="flex flex-col items-center gap-4">
    46→        <div className="w-12 h-12 border-3 border-primary border-t-transparent rounded-full animate-spin" />
    47→        <p className="text-muted-foreground text-sm">Chargement...</p>
    48→      </div>
    49→    </div>
    50→  );
    51→}
    52→
    53→export default function WatchPage() {
    54→  return (
    55→    <Suspense fallback={<WatchLoading />}>
    56→      <WatchContent />

    58→  );

    60→
    61→function WatchContent() {
    62→  const router = useRouter();
    63→  const params = useParams();
    64→  const searchParams = useSearchParams();
    65→  const {
    66→    isPremium,
    67→    watchCount,
    68→    canWatch,
    69→    incrementWatchCount,
    70→    recordWatchSeconds,
    71→    hasWatchedContent,
    72→    BASIC_LIMIT,
    73→    isLoaded,
    74→  } = useChallenge();
    75→  const { addToHistory, updateProgress, getHistoryEntry } = useWatchHistory();
    76→  const progressSaveRef = useRef<ReturnType<typeof setInterval> | null>(null);
    77→
    78→  const mediaType = params.type as 'movie' | 'tv';
    79→  const mediaId = parseInt(params.id as string);
    80→  const urlSeason = searchParams.get('s') ? parseInt(searchParams.get('s')!) : undefined;
    81→  const urlEpisode = searchParams.get('e') ? parseInt(searchParams.get('e')!) : undefined;
    82→
    83→  const isMovie = mediaType === 'movie';
    84→
    85→  const [details, setDetails] = useState<MovieDetails | TVDetails | null>(null);
    86→  const [loading, setLoading] = useState(true);
    87→  const [currentSeason, setCurrentSeason] = useState(urlSeason || 1);
    88→  const [currentEpisode, setCurrentEpisode] = useState(urlEpisode || 1);
    89→  const [episodes, setEpisodes] = useState<Episode[]>([]);
    90→  const [seasonsList, setSeasonsList] = useState<{ number: number; name: string; episodeCount: number }[]>([]);
    91→  const [serverIndex, setServerIndex] = useState(0);
    92→  const [iframeKey, setIframeKey] = useState(0);
    93→  const [isBlocked, setIsBlocked] = useState(false);
    94→  const [watchTimer, setWatchTimer] = useState(0);
    95→  const [subtitlesActive, setSubtitlesActive] = useState(false);
    96→
    97→  // Check access control
    98→  const hasCheckedAccess = useRef(false);
    99→  useEffect(() => {
   100→    if (!isLoaded || hasCheckedAccess.current) return;
   101→    hasCheckedAccess.current = true;
   102→
   103→    if (!canWatch()) {
   104→      setIsBlocked(true);
   105→    } else {
   106→      setIsBlocked(false);
   107→      const allowed = incrementWatchCount();
   108→      if (!allowed) {
   109→        setIsBlocked(true);
   110→      }
   111→    }
   112→  }, [isLoaded]); // eslint-disable-line react-hooks/exhaustive-deps
   113→
   114→  // Track watch time for challenge (Task 2)
   115→  const watchTimerRef = useRef(0);
   116→  useEffect(() => {
   117→    if (isBlocked || !isLoaded) return;
   118→    const interval = setInterval(() => {
   119→      setWatchTimer(prev => prev + 1);
   120→    }, 1000);
   121→    return () => clearInterval(interval);
   122→  }, [isBlocked, isLoaded]);
   123→
   124→  useEffect(() => {
   125→    watchTimerRef.current = watchTimer;
   126→    if (watchTimer > 0) {
   127→      recordWatchSeconds(watchTimer);
   128→    }
   129→  }, [watchTimer, recordWatchSeconds]);
   130→
   131→  // Restore last watched position from history
   132→  useEffect(() => {
   133→    if (!mediaId || !mediaType) return;
   134→    const saved = getHistoryEntry(mediaId, mediaType);
   135→    if (saved) {
   136→      if (!isMovie && saved.season) setCurrentSeason(saved.season);
   137→      if (!isMovie && saved.episode) setCurrentEpisode(saved.episode);
   138→    }
   139→  }, [mediaId, mediaType]); // eslint-disable-line react-hooks/exhaustive-deps
   140→
   141→  // Fetch media details
   142→  useEffect(() => {
   143→    if (!mediaId || !mediaType) {
   144→      router.push('/');
   145→      return;
   146→    }
   147→
   148→    const fetchDetails = async () => {
   149→      setLoading(true);
   150→      try {
   151→        const endpoint = isMovie ? `/movie/${mediaId}` : `/tv/${mediaId}`;
   152→        const data = await fetchTMDB<MovieDetails | TVDetails>(endpoint, { language: 'fr-FR' });
   153→        if (data) {
   154→          setDetails(data);
   155→
   156→          if (!isMovie && data.seasons) {
   157→            const filtered = data.seasons
   158→              .filter((s: { season_number: number }) => s.season_number > 0)
   159→              .map((s: { season_number: number; name: string; episode_count: number }) => ({
   160→                number: s.season_number,
   161→                name: s.name,
   162→                episodeCount: s.episode_count,
   163→              }));
   164→            setSeasonsList(filtered);
   165→          }
   166→        }
   167→      } catch (err) {
   168→        console.error('Failed to fetch details:', err);
   169→      } finally {
   170→        setLoading(false);
   171→      }
   172→    };
   173→
   174→    fetchDetails();
   175→  }, [mediaId, mediaType, isMovie, router]);
   176→
   177→  // Fetch episodes for TV when season changes
   178→  useEffect(() => {
   179→    if (isMovie || !mediaId) return;
   180→
   181→    const fetchEpisodes = async () => {
   182→      try {
   183→        const data: SeasonDetail | null = await fetchTMDB<SeasonDetail>(`/tv/${mediaId}/season/${currentSeason}`, { language: 'fr-FR' });
   184→        if (data) {
   185→          setEpisodes(data.episodes || []);
   186→          if (currentEpisode > (data.episodes?.length || 1)) {
   187→            setCurrentEpisode(1);
   188→          }
   189→        }
   190→      } catch (err) {
   191→        console.error('Failed to fetch episodes:', err);
   192→      }
   193→    };
   194→
   195→    fetchEpisodes();
   196→  }, [mediaId, currentSeason, isMovie, currentEpisode]);
   197→
   198→  // Update URL when season/episode changes (for TV)
   199→  useEffect(() => {
   200→    if (!isMovie && typeof window !== 'undefined') {
   201→      const url = new URL(window.location.href);
   202→      url.searchParams.set('s', String(currentSeason));
   203→      url.searchParams.set('e', String(currentEpisode));
   204→      window.history.replaceState({}, '', url);
   205→    }
   206→  }, [currentSeason, currentEpisode, isMovie]);
   207→
   208→  // Build video URL using API_CONFIG servers
   209→  const getVideoUrl = useCallback(() => {
   210→    const servers = Object.values(API_CONFIG.videoServers);
   211→    const server = servers[serverIndex % servers.length];
   212→
   213→    if (isMovie) {
   214→      return server.movieUrl(mediaId);
   215→    } else {
   216→      return server.tvUrl(mediaId, currentSeason, currentEpisode);
   217→    }
   218→  }, [mediaId, isMovie, currentSeason, currentEpisode, serverIndex]);
   219→
   220→  // Reload iframe when switching server/episode
   221→  const reloadPlayer = () => {
   222→    setIframeKey(k => k + 1);
   223→  };
   224→
   225→  useEffect(() => {
   226→    reloadPlayer();
   227→  }, [serverIndex, currentSeason, currentEpisode]);
   228→
   229→  // Save progress to history periodically and on unmount
   230→  useEffect(() => {
   231→    if (!details || !mediaId || !mediaType || isBlocked) return;
   232→
   233→    const title = getMediaTitle(details);
   234→    addToHistory({
   235→      id: mediaId,
   236→      mediaType,
   237→      title,
   238→      posterPath: details.poster_path || null,
   239→      backdropPath: details.backdrop_path || null,
   240→      overview: details.overview || '',
   241→      progress: 0,
   242→      timestamp: 0,
   243→      duration: isMovie ? ((details as MovieDetails).runtime || 0) * 60 : 0,
   244→      season: !isMovie ? currentSeason : undefined,
   245→      episode: !isMovie ? currentEpisode : undefined,
   246→      episodeTitle: !isMovie ? episodes.find(e => e.episode_number === currentEpisode)?.name : undefined,
   247→      genreIds: details.genre_ids || details.genres?.map((g: { id: number }) => g.id),
   248→      voteAverage: details.vote_average,
   249→      year: getMediaYear(details as Media),
   250→    });
   251→
   252→    // Save progress every 15 seconds
   253→    progressSaveRef.current = setInterval(() => {
   254→      const elapsed = watchTimerRef.current;
   255→      const duration = isMovie ? ((details as MovieDetails).runtime || 0) * 60 : 2400;
   256→      const progress = duration > 0 ? Math.min(Math.round((elapsed / duration) * 100), 100) : 0;
   257→      updateProgress(
   258→        mediaId,
   259→        mediaType,
   260→        progress,
   261→        elapsed,
   262→        duration,
   263→        !isMovie ? currentSeason : undefined,
   264→        !isMovie ? currentEpisode : undefined,
   265→        !isMovie ? episodes.find(e => e.episode_number === currentEpisode)?.name : undefined
   266→      );
   267→    }, 15000);
   268→
   269→    return () => {
   270→      if (progressSaveRef.current) clearInterval(progressSaveRef.current);
   271→      // Final save on unmount
   272→      const elapsed = watchTimerRef.current;
   273→      const duration = isMovie ? ((details as MovieDetails).runtime || 0) * 60 : 2400;
   274→      const progress = duration > 0 ? Math.min(Math.round((elapsed / duration) * 100), 100) : 0;
   275→      updateProgress(
   276→        mediaId,
   277→        mediaType,
   278→        progress,
   279→        elapsed,
   280→        duration,
   281→        !isMovie ? currentSeason : undefined,
   282→        !isMovie ? currentEpisode : undefined,
   283→        !isMovie ? episodes.find(e => e.episode_number === currentEpisode)?.name : undefined
   284→      );
   285→    };
   286→  }, [details, mediaId, mediaType, isBlocked, currentSeason, currentEpisode]); // eslint-disable-line react-hooks/exhaustive-deps
   287→
   288→  // Reset watch timer when switching episodes (TV series)
   289→  useEffect(() => {
   290→    if (!isMovie) {
   291→      setWatchTimer(0);
   292→    }
   293→  }, [currentSeason, currentEpisode]); // eslint-disable-line react-hooks/exhaustive-deps
   294→
   295→  // Also save when episode/season changes
   296→  useEffect(() => {
   297→    if (!details || !mediaId || !mediaType) return;
   298→    const title = getMediaTitle(details);
   299→    const epTitle = !isMovie ? episodes.find(e => e.episode_number === currentEpisode)?.name : undefined;
   300→    updateProgress(
   301→      mediaId,
   302→      mediaType,
   303→      0,
   304→      0,
   305→      isMovie ? ((details as MovieDetails).runtime || 0) * 60 : 2400,
   306→      !isMovie ? currentSeason : undefined,
   307→      !isMovie ? currentEpisode : undefined,
   308→      epTitle
   309→    );
   310→  }, [currentSeason, currentEpisode]); // eslint-disable-line react-hooks/exhaustive-deps
   311→
   312→  const title = details ? getMediaTitle(details) : '';
   313→  const year = details ? getMediaYear(details as Media) : '';
   314→  const backdropUrl = getBackdropUrl(details?.backdrop_path || null, 'large');
   315→  const rating = details?.vote_average?.toFixed(1) || 'N/A';
   316→  const genres = (details as MovieDetails)?.genres || (details as TVDetails)?.genres || [];
   317→  const overview = details?.overview || '';
   318→  const runtime = (details as MovieDetails)?.runtime;
   319→  const formattedRuntime = runtime ? `${Math.floor(runtime / 60)}h ${runtime % 60}min` : null;
   320→  const numberOfSeasons = (details as TVDetails)?.number_of_seasons;
   321→  const tagline = (details as MovieDetails)?.tagline || (details as TVDetails)?.tagline;
   322→
   323→  // Show loading until both TMDB data and challenge state are ready
   324→  if (loading || !isLoaded) {
   325→    return <WatchLoading />;
   326→  }
   327→
   328→  if (!details) {
   329→    return (
   330→      <div className="min-h-screen bg-background flex items-center justify-center px-4">
   331→        <div className="text-center">
   332→          <h1 className="text-xl sm:text-2xl font-bold mb-2">Contenu introuvable</h1>
   333→          <p className="text-muted-foreground mb-4 text-sm">Le contenu demand&eacute; n&apos;existe pas.</p>
   334→          <Link href="/" className="text-primary hover:underline text-sm">Retour &agrave; l&apos;accueil</Link>
   335→        </div>
   336→      </div>
   337→    );
   338→  }
   339→
   340→  // Access blocked overlay
   341→  if (isBlocked) {
   342→    return (
   343→      <div className="min-h-screen bg-background flex items-center justify-center px-4">
   344→        <div className="max-w-md w-full text-center">
   345→          <div className="w-20 h-20 mx-auto mb-6 rounded-2xl bg-gradient-to-br from-primary/20 to-amber-500/20 flex items-center justify-center">
   346→            <svg className="w-10 h-10 text-primary" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   347→              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 15v2m-6 4h12a2 2 0 002-2v-6a2 2 0 00-2-2H6a2 2 0 00-2 2v6a2 2 0 002 2zm10-10V7a4 4 0 00-8 0v4h8z" />
   348→            </svg>
   349→          </div>
   350→          <h1 className="text-2xl sm:text-3xl font-black text-foreground mb-3">Limite atteinte</h1>
   351→          <p className="text-sm text-muted-foreground mb-2">
   352→            Vous avez utilis&eacute; vos {BASIC_LIMIT} contenus gratuits de la version basique.
   353→          </p>
   354→          <p className="text-sm text-muted-foreground mb-6">
   355→            Accomplissez les 3 d&eacute;fis simples pour d&eacute;bloquer l&apos;acc&egrave;s illimit&eacute; &mdash; c&apos;est 100% gratuit !
   356→          </p>
   357→          <Link
   358→            href="/pricing"
   359→            className="inline-flex items-center gap-2 px-6 py-3 rounded-xl bg-gradient-to-r from-primary to-amber-500 text-black font-bold text-sm hover:shadow-lg hover:shadow-primary/25 transition-all hover:scale-105"
   360→          >
   361→            <svg className="w-4 h-4" fill="currentColor" viewBox="0 0 24 24">
   362→              <path d="M5 16L3 5l5.5 5L12 4l3.5 6L21 5l-2 11H5zm0 2h14v2H5v-2z" />
   363→            </svg>
   364→            D&eacute;bloquer l&apos;acc&egrave;s illimit&eacute;
   365→          </Link>
   366→          <div className="mt-4">
   367→            <Link href="/" className="text-xs text-muted-foreground hover:text-primary transition-colors">
   368→              ← Retour &agrave; l&apos;accueil
   369→            </Link>
   370→          </div>
   371→        </div>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  const serverEntries = Object.values(API_CONFIG.videoServers);
   377→
   378→  // Get history entry for progress display
   379→  const historyEntry = getHistoryEntry(mediaId, mediaType);
   380→  const hasProgress = historyEntry && historyEntry.progress > 0 && historyEntry.progress < 95;
   381→
   382→  return (
   383→    <div className="min-h-screen bg-background">
   384→      {/* Top Bar */}
   385→      <div className="fixed top-0 left-0 right-0 z-50 flex items-center justify-between px-3 sm:px-4 h-11 sm:h-12 bg-black/90 backdrop-blur-md border-b border-white/5">
   386→        <div className="flex items-center gap-2 sm:gap-3 min-w-0">
   387→          <Link href="/" className="flex items-center gap-1.5 text-primary font-bold text-xs sm:text-sm flex-shrink-0">
   388→            <svg viewBox="0 0 24 24" className="w-4 h-4 sm:w-5 sm:h-5 fill-primary">
   389→              <path d="M8 5v14l11-7z" />
   390→            </svg>
   391→            <span className="hidden xs:inline">NETPLUS</span>
   392→          </Link>
   393→          {isPremium && (
   394→            <span className="px-1.5 py-0.5 rounded bg-primary/20 text-primary text-[9px] sm:text-[10px] font-bold flex-shrink-0 animate-premium-glow">
   395→              VIP
   396→            </span>
   397→          )}
   398→          <span className="text-white/20 flex-shrink-0">|</span>
   399→          <span className="text-white/70 text-xs sm:text-sm font-medium truncate">{title}</span>
   400→          {!isMovie && (
   401→            <span className="text-primary/60 text-[10px] sm:text-xs flex-shrink-0">
   402→              S{currentSeason}E{currentEpisode}
   403→            </span>
   404→          )}
   405→        </div>
   406→        <button
   407→          onClick={() => router.back()}
   408→          className="text-white/50 hover:text-white text-xs sm:text-sm flex items-center gap-1 transition-colors flex-shrink-0"
   409→        >
   410→          <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   411→            <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
   412→          </svg>
   413→          <span className="hidden sm:inline">Fermer</span>
   414→        </button>
   415→      </div>
   416→
   417→      {/* Video Player with smart server detection + AI Subtitles */}
   418→      <div className="pt-11 sm:pt-12 relative">
   419→        <SmartVideoPlayer
   420→          key={iframeKey}
   421→          src={getVideoUrl()}
   422→          title={title}
   423→          serverIndex={serverIndex}
   424→          totalServers={serverEntries.length}
   425→          onNextServer={() => setServerIndex(i => (i + 1) % serverEntries.length)}
   426→          onRetry={() => setIframeKey(k => k + 1)}
   427→          aspectClass="aspect-video"
   428→        />
   429→        {/* AI Subtitle text overlay on top of video */}
   430→        <SubtitleOverlay
   431→          tmdbId={mediaId}
   432→          mediaType={isMovie ? 'movie' : 'tv'}
   433→          season={!isMovie ? currentSeason : undefined}
   434→          episode={!isMovie ? currentEpisode : undefined}
   435→          currentTime={watchTimer}
   436→          isActive={subtitlesActive}
   437→          onToggle={() => setSubtitlesActive(prev => !prev)}
   438→          overlayOnly
   439→        />
   440→      </div>
   441→
   442→      {/* Resume indicator */}
   443→      {hasProgress && (
   444→        <div className="px-3 sm:px-6 pt-3">
   445→          <div className="flex items-center gap-2 px-3 py-2 rounded-lg bg-primary/5 border border-primary/10">
   446→            <svg className="w-3.5 h-3.5 text-primary flex-shrink-0" fill="none" stroke="currentColor" viewBox="0 0 24 24">
   447→              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 8v4l3 3m6-3a9 9 0 11-18 0 9 9 0 0118 0z" />
   448→            </svg>
   449→            <span className="text-xs text-primary/70">
   450→              Reprise &agrave; {historyEntry.progress}% &mdash; {historyEntry.timestamp > 60 ? `${Math.floor(historyEntry.timestamp / 60)}min` : `${historyEntry.timestamp}s`}
   451→            </span>
   452→            <div className="flex-1 h-1 bg-primary/10 rounded-full overflow-hidden ml-1">
   453→              <div className="h-full bg-primary rounded-full" style={{ width: `${historyEntry.progress}%` }} />
   454→            </div>
   455→          </div>
   456→        </div>
   457→      )}
   458→
   459→      {/* Content Below Player */}
   460→      <div className="px-3 sm:px-6 py-4 sm:py-8">
   461→        {/* Title & Meta */}
   462→        <div className="mb-4 sm:mb-6">
   463→          {tagline && (
   464→            <p className="text-primary/70 text-[11px] sm:text-sm italic mb-0.5">{tagline}</p>
   465→          )}
   466→          <h1 className="text-lg sm:text-2xl lg:text-3xl font-black text-foreground mb-2 leading-tight">{title}</h1>
   467→
   468→          {/* Meta row */}
   469→          <div className="flex flex-wrap items-center gap-x-3 gap-y-1 text-xs sm:text-sm text-muted-foreground mb-3">
   470→            <span className="flex items-center gap-1">
   471→              <svg className="w-3.5 h-3.5 text-primary fill-primary" viewBox="0 0 24 24">
   472→                <path d="M12 2l3.09 6.26L22 9.27l-5 4.87 1.18 6.88L12 17.77l-6.18 3.25L7 14.14 2 9.27l6.91-1.01L12 2z" />
   473→              </svg>
   474→              <span className="font-semibold">{rating}</span>
   475→            </span>
   476→            {year && <span>{year}</span>}
   477→            {formattedRuntime && <span>{formattedRuntime}</span>}
   478→            {numberOfSeasons && <span>{numberOfSeasons} saison{numberOfSeasons > 1 ? 's' : ''}</span>}
   479→            <span className="px-1.5 py-0.5 rounded bg-primary/20 text-primary text-[10px] sm:text-xs font-bold">
   480→              {isMovie ? 'FILM' : 'SÉRIE'}
   481→            </span>
   482→          </div>
   483→
   484→          {/* Genres */}
   485→          {genres.length > 0 && (
   486→            <div className="flex flex-wrap gap-1.5 sm:gap-2 mb-3">
   487→              {genres.map(g => (
   488→                <span key={g.id} className="px-2 sm:px-3 py-0.5 sm:py-1 rounded-full bg-white/5 text-[10px] sm:text-xs text-white/60 border border-white/10">
   489→                  {g.name}
   490→                </span>
   491→              ))}
   492→            </div>
   493→          )}
   494→
   495→          {/* Overview */}
   496→          <p className="text-xs sm:text-sm text-muted-foreground leading-relaxed line-clamp-3 sm:line-clamp-none">{overview}</p>
   497→        </div>
   498→
   499→        {/* Server Switch + AI Subtitles */}
