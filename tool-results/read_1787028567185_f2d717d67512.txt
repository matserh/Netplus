    57→// Fetch helper
    58→const fetchTMDB = async <T,>(endpoint: string): Promise<T | null> => {
    59→  try {
    60→    const res = await fetch(
    61→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`
    62→    );
    63→    return res.ok ? await res.json() : null;
    64→  } catch {
    65→    return null;
    66→  }
    67→};

    69→// Promotional banner
    70→function PromoBanner() {
    71→  const [items, setItems] = useState<Media[]>([]);
    72→  const [currentIndex, setCurrentIndex] = useState(0);
    73→
    74→  useEffect(() => {
    75→    const load = async () => {
    76→      const data = await fetchTMDB<TMDBResponse<Media>>('/movie/now_playing?region=FR');
    77→      if (data?.results) {
    78→        setItems(data.results.filter(m => m.backdrop_path).slice(0, 5));
    79→      }
    80→    };
    81→    load();
    82→  }, []);
    83→
    84→  useEffect(() => {
    85→    if (items.length <= 1) return;
    86→    const interval = setInterval(() => {
    87→      setCurrentIndex(prev => (prev + 1) % items.length);
    88→    }, 6000);
    89→    return () => clearInterval(interval);
    90→  }, [items.length]);
    91→
    92→  if (items.length === 0) return null;
    93→
    94→  const current = items[currentIndex];
    95→  const title = current.title || current.name || '';
    96→
    97→  return (
    98→    <div className="relative w-full h-[35vh] sm:h-[40vh] overflow-hidden">
    99→      <div className="absolute inset-0">
   100→        <img
   101→          src={`${API_CONFIG.tmdb.imageUrl}/original${current.backdrop_path}`}
   102→          alt={title}
   103→          className="w-full h-full object-cover transition-all duration-1000"
   104→          key={currentIndex}
   105→        />
   106→      </div>
   107→      <div className="absolute inset-0 bg-gradient-to-b from-background/30 via-background/10 to-background" />
   108→      <div className="absolute inset-0 bg-gradient-to-r from-background/50 via-transparent to-transparent" />
   109→
   110→      {/* Dots */}
   111→      {items.length > 1 && (
   112→        <div className="absolute bottom-3 right-6 flex items-center gap-1.5">
   113→          {items.map((_, i) => (
   114→            <button
   115→              key={i}
   116→              onClick={() => setCurrentIndex(i)}
   117→              className={`w-1.5 h-1.5 rounded-full transition-all duration-300 ${
   118→                i === currentIndex ? 'bg-primary w-4' : 'bg-white/30 hover:bg-white/50'
   119→              }`}
   120→            />
   121→          ))}
   122→        </div>
   123→      )}
   124→    </div>
   125→  );
   126→}

   461→          <p className="text-sm text-primary mb-2 animate-fade-in">
   462→            Cliquez sur un profil pour le modifier
   463→          </p>
   464→        )}
   465→
   466→        {/* Profile cards */}
   467→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-6 sm:gap-8 md:gap-10 justify-items-center my-8 sm:my-10">
   468→          {profiles.map((profile, i) => {
   469→            const imgSrc = PROFILE_IMAGES[profile.type];
   470→            const gradient = PROFILE_COLORS[profile.type];
   471→            const desc = PROFILE_DESCRIPTIONS[profile.type];
   472→            const isActive = currentProfile?.type === profile.type;
   473→            const [hovered, setHovered] = useState(false);
   474→            const [visible, setVisible] = useState(false);
   475→            const [imgLoaded, setImgLoaded] = useState(false);
   476→
   477→            useEffect(() => {
   478→              const timer = setTimeout(() => setVisible(true), 150 + i * 120);
   479→              return () => clearTimeout(timer);
   480→            }, []);
   481→
   482→            return (
   483→              <button
   484→                key={profile.type}
   485→                onClick={() => handleSelect(profile)}
   486→                onMouseEnter={() => setHovered(true)}
   487→                onMouseLeave={() => setHovered(false)}
   488→                className={`flex flex-col items-center gap-3 group transition-all duration-500 ${
   489→                  visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'
   490→                }`}
   491→              >
   492→                {/* Avatar */}
   493→                <div
   494→                  className={`relative w-28 h-28 sm:w-32 sm:h-32 md:w-36 md:h-36 rounded-xl overflow-hidden
   495→                    ring-2 ${isActive ? 'ring-primary ring-offset-2 ring-offset-background' : hovered ? 'ring-white/60 ring-offset-2 ring-offset-background' : 'ring-transparent'}
   496→                    transition-all duration-300 ${hovered ? 'scale-105' : 'scale-100'}
   497→                    shadow-lg bg-muted`}
   498→                >
   499→                  {!imgLoaded && <div className="absolute inset-0 bg-muted animate-pulse" />}
   500→                  <img
   501→                    src={imgSrc}
   502→                    alt={profile.name}
   503→                    className={`w-full h-full object-cover transition-all duration-300 ${
   504→                      imgLoaded ? 'opacity-100' : 'opacity-0'
   505→                    } group-hover:scale-110`}
   506→                    onLoad={() => setImgLoaded(true)}
   507→                  />
   508→
   509→                  {/* Active badge */}
   510→                  {isActive && !managing && (
   511→                    <div className="absolute bottom-1 right-1 w-6 h-6 rounded-full bg-primary flex items-center justify-center shadow-md">
   512→                      <Check className="w-3.5 h-3.5 text-black" />
   513→                    </div>
   514→                  )}
   515→
   516→                  {/* Managing overlay */}
   517→                  {managing && (
   518→                    <div className={`absolute inset-0 bg-black/50 flex items-center justify-center transition-opacity duration-200 ${
   519→                      hovered ? 'opacity-100' : 'opacity-0'
   520→                    }`}>
   521→                      <Pencil className="w-8 h-8 text-white" />
   522→                    </div>
   523→                  )}
   524→
   525→                  {/* Hover overlay */}
   526→                  {!managing && !isActive && hovered && (
   527→                    <div className="absolute inset-0 bg-black/30 flex items-center justify-center animate-fade-in">
   528→                      <div className="w-12 h-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center">
   529→                        <Check className="w-7 h-7 text-white" />
   530→                      </div>
   531→                    </div>
   532→                  )}
   533→                </div>
   534→
   535→                {/* Name */}
   536→                <span className={`text-sm sm:text-base font-semibold transition-colors duration-200 ${
   537→                  isActive ? 'text-primary' : hovered ? 'text-foreground' : 'text-foreground/60'
   538→                }`}>
   539→                  {profile.name}
   540→                </span>
   541→
   542→                {/* Description */}
   543→                <span className="text-[11px] text-muted-foreground/40 max-w-[120px] text-center">
   544→                  {desc}
   545→                </span>
   546→              </button>
   547→            );
   548→          })}
   549→        </div>
   550→
   551→        {/* Managing toggle */}
   552→        {managing && (
   553→          <button
   554→            onClick={() => setManaging(false)}
   555→            className="flex items-center gap-2 px-6 py-2.5 rounded-full bg-primary text-black font-semibold hover:bg-primary/90 transition-all text-sm"
   556→          >
   557→            Terminé
   558→          </button>
   559→        )}
   560→      </div>
   561→
   562→      {/* Bottom */}
   563→      <div className="py-3 text-center">
   564→        <p className="text-[10px] text-muted-foreground/20">
   565→          © 2026 Netplus · Aeronlabs + IAgen
   566→        </p>
   567→      </div>
   568→    </div>
   569→  );
   570→}
