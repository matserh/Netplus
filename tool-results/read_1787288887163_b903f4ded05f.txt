    53→    53→  try {
    54→    54→    const res = await fetch(
    55→    55→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`
    56→    56→    );
    57→    57→    return res.ok ? await res.json() : null;
    58→    58→  } catch {
    59→    59→    return null;
    60→    60→  }

    63→    63→// Promotional banner
    64→    64→function PromoBanner() {
    65→    65→  const [items, setItems] = useState<Media[]>([]);
    66→    66→  const [currentIndex, setCurrentIndex] = useState(0);
    67→    67→
    68→    68→  useEffect(() => {
    69→    69→    const load = async () => {
    70→    70→      const data = await fetchTMDB<TMDBResponse<Media>>('/movie/now_playing?region=FR');
    71→    71→      if (data?.results) {
    72→    72→        setItems(data.results.filter(m => m.backdrop_path).slice(0, 5));
    73→    73→      }
    74→    74→    };
    75→    75→    load();
    76→    76→  }, []);
    77→    77→
    78→    78→  useEffect(() => {
    79→    79→    if (items.length <= 1) return;
    80→    80→    const interval = setInterval(() => {
    81→    81→      setCurrentIndex(prev => (prev + 1) % items.length);
    82→    82→    }, 6000);
    83→    83→    return () => clearInterval(interval);
    84→    84→  }, [items.length]);
    85→    85→
    86→    86→  if (items.length === 0) return null;
    87→    87→
    88→    88→  const current = items[currentIndex];
    89→    89→  const title = current.title || current.name || '';
    90→    90→
    91→    91→  return (
    92→    92→    <div className="relative w-full h-[35vh] sm:h-[40vh] overflow-hidden">
    93→    93→      <div className="absolute inset-0">
    94→    94→        <img
    95→    95→          src={`${API_CONFIG.tmdb.imageUrl}/original${current.backdrop_path}`}
    96→    96→          alt={title}
    97→    97→          className="w-full h-full object-cover transition-all duration-1000"
    98→    98→          key={currentIndex}
    99→    99→        />
   100→   100→      </div>
   101→   101→      <div className="absolute inset-0 bg-gradient-to-b from-background/30 via-background/10 to-background" />
   102→   102→      <div className="absolute inset-0 bg-gradient-to-r from-background/50 via-transparent to-transparent" />
   103→   103→
   104→   104→      {/* Dots */}
   105→   105→      {items.length > 1 && (
   106→   106→        <div className="absolute bottom-3 right-6 flex items-center gap-1.5">
   107→   107→          {items.map((_, i) => (
   108→   108→            <button
   109→   109→              key={i}
   110→   110→              onClick={() => setCurrentIndex(i)}
   111→   111→              className={`w-1.5 h-1.5 rounded-full transition-all duration-300 ${
   112→   112→                i === currentIndex ? 'bg-primary w-4' : 'bg-white/30 hover:bg-white/50'
   113→   113→              }`}
   114→   114→            />
   115→   115→          ))}
   116→   116→        </div>
   117→   117→      )}
   118→   118→    </div>
   119→   119→  );
   120→   120→}
