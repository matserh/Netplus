    51→// Fetch helper
    52→const fetchTMDB = async <T,>(endpoint: string): Promise<T | null> => {
    53→  try {
    54→    const res = await fetch(
    55→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`
    56→    );
    57→    return res.ok ? await res.json() : null;
    58→  } catch {
    59→    return null;
    60→  }
    61→};

    63→// Promotional banner
    64→function PromoBanner() {
    65→  const [items, setItems] = useState<Media[]>([]);
    66→  const [currentIndex, setCurrentIndex] = useState(0);
    67→
    68→  useEffect(() => {
    69→    const load = async () => {
    70→      const data = await fetchTMDB<TMDBResponse<Media>>('/movie/now_playing?region=FR');
    71→      if (data?.results) {
    72→        setItems(data.results.filter(m => m.backdrop_path).slice(0, 5));
    73→      }
    74→    };
    75→    load();
    76→  }, []);
    77→
    78→  useEffect(() => {
    79→    if (items.length <= 1) return;
    80→    const interval = setInterval(() => {
    81→      setCurrentIndex(prev => (prev + 1) % items.length);
    82→    }, 6000);
    83→    return () => clearInterval(interval);
    84→  }, [items.length]);
    85→
    86→  if (items.length === 0) return null;
    87→
    88→  const current = items[currentIndex];
    89→  const title = current.title || current.name || '';
    90→
    91→  return (
    92→    <div className="relative w-full h-[35vh] sm:h-[40vh] overflow-hidden">
    93→      <div className="absolute inset-0">
    94→        <img
    95→          src={`${API_CONFIG.tmdb.imageUrl}/original${current.backdrop_path}`}
    96→          alt={title}
    97→          className="w-full h-full object-cover transition-all duration-1000"
    98→          key={currentIndex}
    99→        />
   100→      </div>
   101→      <div className="absolute inset-0 bg-gradient-to-b from-background/30 via-background/10 to-background" />
   102→      <div className="absolute inset-0 bg-gradient-to-r from-background/50 via-transparent to-transparent" />
   103→
   104→      {/* Dots */}
   105→      {items.length > 1 && (
   106→        <div className="absolute bottom-3 right-6 flex items-center gap-1.5">
   107→          {items.map((_, i) => (
   108→            <button
   109→              key={i}
   110→              onClick={() => setCurrentIndex(i)}
   111→              className={`w-1.5 h-1.5 rounded-full transition-all duration-300 ${
   112→                i === currentIndex ? 'bg-primary w-4' : 'bg-white/30 hover:bg-white/50'
   113→              }`}
   114→            />
   115→          ))}
   116→        </div>
   117→      )}
   118→    </div>
   119→  );
   120→}

   122→function ProfileCard({ profile, index, isActive, isManaging, onSelect }: {
   123→  profile: UserProfile;
   124→  index: number;
   125→  isActive: boolean;
   126→  isManaging: boolean;
   127→  onSelect: (p: UserProfile) => void;
   128→}) {
   129→  const imgSrc = PROFILE_IMAGES[profile.type];
   130→  const desc = PROFILE_DESCRIPTIONS[profile.type];
   131→  const [hovered, setHovered] = useState(false);
   132→  const [visible, setVisible] = useState(false);
   133→  const [imgLoaded, setImgLoaded] = useState(false);
   134→
   135→  useEffect(() => {
   136→    const timer = setTimeout(() => setVisible(true), 150 + index * 120);
   137→    return () => clearTimeout(timer);
   138→  }, [index]);
   139→
   140→  return (
   141→    <button
   142→      onClick={() => onSelect(profile)}
   143→      onMouseEnter={() => setHovered(true)}
   144→      onMouseLeave={() => setHovered(false)}
   145→      className={`flex flex-col items-center gap-3 group transition-all duration-500 ${
   146→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'
   147→      }`}
   148→    >
   149→      <div
   150→        className={`relative w-28 h-28 sm:w-32 sm:h-32 md:w-36 md:h-36 rounded-xl overflow-hidden
   151→          ring-2 ${isActive ? 'ring-primary ring-offset-2 ring-offset-background' : hovered ? 'ring-white/60 ring-offset-2 ring-offset-background' : 'ring-transparent'}
   152→          transition-all duration-300 ${hovered ? 'scale-105' : 'scale-100'}
   153→          shadow-lg bg-muted`}
   154→      >
   155→        {!imgLoaded && <div className="absolute inset-0 bg-muted animate-pulse" />}
   156→        <img
   157→          src={imgSrc}
   158→          alt={profile.name}
   159→          className={`w-full h-full object-cover transition-all duration-300 ${
   160→            imgLoaded ? 'opacity-100' : 'opacity-0'
   161→          } group-hover:scale-110`}
   162→          onLoad={() => setImgLoaded(true)}
   163→        />
   164→        {isActive && !isManaging && (
   165→          <div className="absolute bottom-1 right-1 w-6 h-6 rounded-full bg-primary flex items-center justify-center shadow-md">
   166→            <Check className="w-3.5 h-3.5 text-black" />
   167→          </div>
   168→        )}
   169→        {isManaging && (
   170→          <div className={`absolute inset-0 bg-black/50 flex items-center justify-center transition-opacity duration-200 ${
   171→            hovered ? 'opacity-100' : 'opacity-0'
   172→          }`}>
   173→            <Pencil className="w-8 h-8 text-white" />
   174→          </div>
   175→        )}
   176→        {!isManaging && !isActive && hovered && (
   177→          <div className="absolute inset-0 bg-black/30 flex items-center justify-center animate-fade-in">
   178→            <div className="w-12 h-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center">
   179→              <Check className="w-7 h-7 text-white" />
   180→            </div>
   181→          </div>
   182→        )}
   183→      </div>
   184→      <span className={`text-sm sm:text-base font-semibold transition-colors duration-200 ${
   185→        isActive ? 'text-primary' : hovered ? 'text-foreground' : 'text-foreground/60'
   186→      }`}>
   187→        {profile.name}
   188→      </span>
   189→      <span className="text-[11px] text-muted-foreground/40 max-w-[120px] text-center">
   190→        {desc}
   191→      </span>
   192→    </button>
   193→  );
   194→}
