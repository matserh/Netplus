    34→function ConfettiEffect() {
    35→  const [particles, setParticles] = useState<Array<{ id: number; x: number; y: number; color: string; delay: number; size: number }>>([]);
    36→
    37→  useEffect(() => {
    38→    const colors = ['#FFD700', '#FFA500', '#FF6347', '#FF4500', '#FFD700', '#DAA520'];
    39→    const p = Array.from({ length: 40 }, (_, i) => ({
    40→      id: i,
    41→      x: Math.random() * 100,
    42→      y: -10 - Math.random() * 20,
    43→      color: colors[Math.floor(Math.random() * colors.length)],
    44→      delay: Math.random() * 2,
    45→      size: 4 + Math.random() * 8,
    46→    }));
    47→    setParticles(p);
    48→  }, []);
    49→
    50→  if (particles.length === 0) return null;
    51→
    52→  return (
    53→    <div className="fixed inset-0 pointer-events-none z-50 overflow-hidden">
    54→      {particles.map(p => (
    55→        <div
    56→          key={p.id}
    57→          className="absolute animate-confetti-fall"
    58→          style={{
    59→            left: `${p.x}%`,
    60→            top: `${p.y}%`,
    61→            width: p.size,
    62→            height: p.size,
    63→            backgroundColor: p.color,
    64→            borderRadius: Math.random() > 0.5 ? '50%' : '2px',
    65→            animationDelay: `${p.delay}s`,
    66→          }}
    67→        />
    68→      ))}
    69→    </div>
    70→  );
    71→}
