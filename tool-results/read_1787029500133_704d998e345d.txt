    20→    20→const fetchTMDB = async <T,>(endpoint: string, timeoutMs: number = 10000): Promise<T | null> => {
    21→    21→  try {
    22→    22→    const controller = new AbortController();
    23→    23→    const tid = setTimeout(() => controller.abort(), timeoutMs);
    24→    24→    const res = await fetch(
    25→    25→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`,
    26→    26→      { signal: controller.signal }
    27→    27→    );
    28→    28→    clearTimeout(tid);
    29→    29→    return res.ok ? await res.json() : null;
    30→    30→  } catch { return null; }
    31→    31→};

    95→    95→function formatRelativeDate(dateStr: string): string {
    96→    96→  const diff = Date.now() - new Date(dateStr).getTime();
    97→    97→  const min = Math.floor(diff / 60000);
    98→    98→  if (min < 1) return "À l'instant";
    99→    99→  if (min < 60) return `Il y a ${min}min`;
   100→   100→  const h = Math.floor(min / 60);
   101→   101→  if (h < 24) return `Il y a ${h}h`;
   102→   102→  const d = Math.floor(h / 24);
   103→   103→  if (d < 7) return `Il y a ${d}j`;
   104→   104→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' });
   105→   105→}

   129→   129→function SmartShortPlayer({ src, title, onSwitchServer, serverIndex, totalServers }: {
   130→   130→  src: string; title: string; onSwitchServer: () => void; serverIndex: number; totalServers: number;
   131→   131→}) {
   132→   132→  const [state, setState] = useState<'loading' | 'ready' | 'error' | 'timeout'>('loading');
   133→   133→  const [failedServers, setFailedServers] = useState<Set<number>>(new Set());
   134→   134→  const [autoFallbackAttempted, setAutoFallbackAttempted] = useState(false);
   135→   135→  const [allFailed, setAllFailed] = useState(false);
   136→   136→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   137→   137→
   138→   138→  useEffect(() => {
   139→   139→    setState('loading');
   140→   140→    setAutoFallbackAttempted(false);
   141→   141→    if (timerRef.current) clearTimeout(timerRef.current);
   142→   142→    timerRef.current = setTimeout(() => {
   143→   143→      setState(prev => prev === 'loading' ? 'timeout' : prev);
   144→   144→      setFailedServers(prev => new Set(prev).add(serverIndex));
   145→   145→    }, 15000);
   146→   146→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   147→   147→  }, [src, serverIndex]);
   148→   148→
   149→   149→  useEffect(() => {
   150→   150→    if ((state === 'error' || state === 'timeout') && !autoFallbackAttempted) {
   151→   151→      setAutoFallbackAttempted(true);
   152→   152→      const newFailed = new Set(failedServers);
   153→   153→      newFailed.add(serverIndex);
   154→   154→      if (newFailed.size >= totalServers) { setAllFailed(true); return; }
   155→   155→      const timer = setTimeout(() => onSwitchServer(), 1500);
   156→   156→      return () => clearTimeout(timer);
   157→   157→    }
   158→   158→  }, [state, autoFallbackAttempted, failedServers, serverIndex, totalServers, onSwitchServer]);
   159→   159→
   160→   160→  const handleLoad = useCallback(() => {
   161→   161→    if (timerRef.current) clearTimeout(timerRef.current);
   162→   162→    setState('ready');
   163→   163→    setFailedServers(prev => { const n = new Set(prev); n.delete(serverIndex); return n; });
   164→   164→  }, [serverIndex]);
   165→   165→
   166→   166→  const handleError = useCallback(() => {
   167→   167→    if (timerRef.current) clearTimeout(timerRef.current);
   168→   168→    setState('error');
   169→   169→    setFailedServers(prev => new Set(prev).add(serverIndex));
   170→   170→  }, [serverIndex]);
   171→   171→
   172→   172→  const showOverlay = state === 'error' || state === 'timeout';
   173→   173→  const serverName = VIDEO_SERVERS[serverIndex]?.name || `Serveur ${serverIndex + 1}`;
   174→   174→  const available = totalServers - failedServers.size;
   175→   175→  const isAutoFallingBack = showOverlay && !allFailed && !autoFallbackAttempted;
   176→   176→
   177→   177→  return (
   178→   178→    <div className="absolute inset-0 flex items-center justify-center">
   179→   179→      <iframe
   180→   180→        src={src}
   181→   181→        className={cn('w-full h-full transition-opacity duration-500', showOverlay ? 'opacity-0' : 'opacity-100')}
   182→   182→        style={{ aspectRatio: '16/9', minHeight: '100%', maxWidth: '177.78%' }}
   183→   183→        allowFullScreen
   184→   184→        allow="autoplay; fullscreen; picture-in-picture"
   185→   185→        referrerPolicy="origin"
   186→   186→        title={title}
   187→   187→        onLoad={handleLoad}
   188→   188→        onError={handleError}
   189→   189→      />
   190→   190→      {state === 'loading' && !showOverlay && (
   191→   191→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-black/60">
   192→   192→          <div className="flex flex-col items-center gap-2">
   193→   193→            <div className="w-8 h-8 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   194→   194→            <span className="text-[11px] text-white/40">{serverName}</span>
   195→   195→          </div>
   196→   196→        </div>
   197→   197→      )}
   198→   198→      {isAutoFallingBack && (
   199→   199→        <div className="absolute inset-0 flex items-center justify-center z-15 bg-black/80">
   200→   200→          <div className="flex flex-col items-center gap-2 animate-pulse">
   201→   201→            <Wifi className="w-8 h-8 text-primary" />
   202→   202→            <p className="text-xs text-white/50">Serveur suivant...</p>
   203→   203→          </div>
   204→   204→        </div>
   205→   205→      )}
   206→   206→      {showOverlay && allFailed && (
   207→   207→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   208→   208→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[280px]">
   209→   209→            <WifiOff className="w-10 h-10 text-red-400" />
   210→   210→            <p className="text-sm font-bold text-white">Contenu indisponible</p>
   211→   211→            <p className="text-xs text-white/40">Aucun serveur ne peut charger ce contenu.</p>
   212→   212→            <button onClick={() => { setAllFailed(false); setFailedServers(new Set()); onSwitchServer(); }}
   213→   213→              className="flex items-center gap-2 px-5 py-2.5 rounded-full bg-primary text-black text-sm font-bold active:scale-95 transition-transform">
   214→   214→              <RefreshCw className="w-4 h-4" /> Réessayer
   215→   215→            </button>
   216→   216→          </div>
   217→   217→        </div>
   218→   218→      )}
   219→   219→      {showOverlay && !allFailed && !isAutoFallingBack && (
   220→   220→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   221→   221→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[280px]">
   222→   222→            <AlertTriangle className="w-10 h-10 text-amber-400" />
   223→   223→            <p className="text-sm font-bold text-white">Serveur indisponible</p>
   224→   224→            <p className="text-xs text-white/40">{serverName} ne répond pas</p>
   225→   225→            <p className="text-xs text-primary/60">{available} serveur{available > 1 ? 's' : ''} restant{available > 1 ? 's' : ''}</p>
   226→   226→            <button onClick={onSwitchServer}
   227→   227→              className="flex items-center gap-2 px-5 py-2.5 rounded-full bg-primary text-black text-sm font-bold active:scale-95 transition-transform">
   228→   228→              <SkipForward className="w-4 h-4" /> Serveur suivant
   229→   229→            </button>
   230→   230→          </div>
   231→   231→        </div>
   232→   232→      )}
   233→   233→    </div>
   234→   234→  );
   235→   235→}

   246→   246→// ─── Load shorts ───
   247→   247→async function loadShortsFromEndpoint(
   248→   248→  endpoint: string, mediaType: 'movie' | 'tv', genreLabel?: string, maxItems: number = 8, seenIds: Set<string> = new Set()
   249→   249→): Promise<ShortItem[]> {
   250→   250→  const data = await fetchTMDB<TMDBResponse<Media>>(endpoint);
   251→   251→  if (!data?.results) return [];
   252→   252→  const items: ShortItem[] = [];
   253→   253→  for (const item of data.results) {
   254→   254→    if (items.length >= maxItems) break;
   255→   255→    const dedupKey = `${item.id}-${mediaType}`;
   256→   256→    if (seenIds.has(dedupKey)) continue;
   257→   257→    seenIds.add(dedupKey);
   258→   258→    if (mediaType === 'tv') {
   259→   259→      const detail = await fetchTMDB<any>(`/tv/${item.id}`);
   260→   260→      if (detail?.seasons) {
   261→   261→        const firstRealSeason = detail.seasons.find((s: any) => s.season_number > 0);
   262→   262→        if (firstRealSeason) {
   263→   263→          const seasonData = await fetchTMDB<any>(`/tv/${item.id}/season/${firstRealSeason.season_number}`);
   264→   264→          if (seasonData?.episodes?.[0]) {
   265→   265→            const ep = seasonData.episodes[0];
   266→   266→            items.push({ id: item.id, mediaType: 'tv', title: item.name || item.original_name || '', overview: ep.overview || item.overview || '', posterPath: item.poster_path, backdropPath: ep.still_path || item.backdrop_path, voteAverage: item.vote_average, genreLabel, season: firstRealSeason.season_number, episode: ep.episode_number, episodeName: ep.name });
   267→   267→          }
   268→   268→        }
   269→   269→      }
   270→   270→    } else {
   271→   271→      items.push({ id: item.id, mediaType: 'movie', title: item.title || item.original_title || '', overview: item.overview || '', posterPath: item.poster_path, backdropPath: item.backdrop_path, voteAverage: item.vote_average, genreLabel });
   272→   272→    }
   273→   273→  }
   274→   274→  return items;
   275→   275→}

   278→   278→function LikeButton({ mediaId, mediaType, segment, season, likeTriggerRef }: { mediaId: number; mediaType: string; segment: number; season?: number | null; likeTriggerRef?: React.MutableRefObject<(() => void) | null> }) {
   279→   279→  const [data, setData] = useState<LikeData>({ count: 0, liked: false });
   280→   280→  const [animating, setAnimating] = useState(false);
   281→   281→  const [justLiked, setJustLiked] = useState(false);
   282→   282→  const [isToggling, setIsToggling] = useState(false);
   283→   283→  const { data: session } = useSession();
   284→   284→  const normalizedSeason = season || 0;
   285→   285→
   286→   286→  useEffect(() => {
   287→   287→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   288→   288→    fetch(`/api/shorts/like?${params}`)
   289→   289→      .then(r => r.ok ? r.json() : null)
   290→   290→      .then(d => { if (d?.count !== undefined) setData(d); })
   291→   291→      .catch(() => {});
   292→   292→  }, [mediaId, mediaType, segment, normalizedSeason]);
   293→   293→
   294→   294→  const toggleLike = async () => {
   295→   295→    if (!session || isToggling) return;
   296→   296→    setIsToggling(true);
   297→   297→    const willLike = !data.liked;
   298→   298→    setJustLiked(willLike);
   299→   299→    setAnimating(true);
   300→   300→    setData(prev => ({ count: prev.count + (willLike ? 1 : -1), liked: willLike }));
   301→   301→    try {
   302→   302→      const res = await fetch('/api/shorts/like', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason }) });
   303→   303→      if (!res.ok) { setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike })); }
   304→   304→      else { const d = await res.json(); if (d?.liked !== undefined) setData(prev => ({ count: d.count ?? prev.count, liked: d.liked })); else if (d?.error) setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike })); }
   305→   305→    } catch { setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike })); }
   306→   306→    setTimeout(() => { setAnimating(false); setJustLiked(false); setIsToggling(false); }, 600);
   307→   307→  };
   308→   308→
   309→   309→  useEffect(() => { if (likeTriggerRef) likeTriggerRef.current = toggleLike; });
   310→   310→
   311→   311→  return (
   312→   312→    <button onClick={toggleLike} className="flex flex-col items-center gap-1 group active:scale-90 transition-transform">
   313→   313→      <div className="relative">
   314→   314→        <Heart className={cn('w-8 h-8 transition-all duration-300', animating && justLiked && 'scale-125', data.liked ? 'text-red-500 fill-red-500 drop-shadow-[0_0_8px_rgba(239,68,68,0.4)]' : 'text-white group-hover:text-white/90')} strokeWidth={data.liked ? 0 : 1.5} />
   315→   315→        {animating && (
   316→   316→          <div className="absolute inset-0 flex items-center justify-center">
   317→   317→            <div className={cn('w-14 h-14 rounded-full animate-ping-once', justLiked ? 'bg-red-500/25' : 'bg-white/10')} />
   318→   318→          </div>
   319→   319→        )}
   320→   320→      </div>
   321→   321→      <span className={cn('text-[11px] font-semibold transition-colors', data.liked ? 'text-red-400' : 'text-white/60')}>
   322→   322→        {data.count > 0 ? formatCount(data.count) : 'J\'aime'}
   323→   323→      </span>
   324→   324→    </button>
   325→   325→  );
   326→   326→}

   713→   713→          <div className="grid grid-cols-3 sm:grid-cols-4 gap-2.5">
   714→   714→            {results.map(item => (
   715→   715→              <button key={`${item.id}-${item.mediaType}`} onClick={() => { handleClose(); onResultClick(item); }}

   720→   720→                <div className="p-2"><p className="text-[11px] font-semibold text-white/80 truncate">{item.title}</p><span className="text-[9px] text-white/30">{item.mediaType === 'movie' ? 'Film' : 'Série'}</span></div>
   721→   721→              </button>
   722→   722→            ))}
   723→   723→          </div>
   724→   724→        </div>
   725→   725→      </div>
   726→   726→    </div>
   727→   727→  );
   728→   728→}

   731→   731→function getProfileGenreParams(profileType: string | undefined): string {
   732→   732→  switch (profileType) {
   733→   733→    case 'JEUNESSE': return 'with_genres=16|10751|14&without_genres=27|53|80';
   734→   734→    case 'FRENESIE': return 'with_genres=28|18|27|14|878';
   735→   735→    case 'NOCTURNE': default: return '';
   736→   736→  }

   739→   739→function withPage(endpoint: string, page: number): string {
   740→   740→  return endpoint.includes('?') ? `${endpoint}&page=${page}` : `${endpoint}?page=${page}`;
   741→   741→}
