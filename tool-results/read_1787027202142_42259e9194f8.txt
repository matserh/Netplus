    20→    20→const fetchTMDB = async <T,>(endpoint: string, timeoutMs: number = 10000): Promise<T | null> => {
    21→    21→  try {
    22→    22→    const controller = new AbortController();
    23→    23→    const tid = setTimeout(() => controller.abort(), timeoutMs);
    24→    24→    const res = await fetch(
    25→    25→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`,
    26→    26→      { signal: controller.signal }
    27→    27→    );
    28→    28→    clearTimeout(tid);
    29→    29→    return res.ok ? await res.json() : null;
    30→    30→  } catch { return null; }
    31→    31→};

    98→    98→function formatTimeShort(seconds: number): string {
    99→    99→  const m = Math.floor(seconds / 60);
   100→   100→  const s = seconds % 60;
   101→   101→  return `${m}:${s.toString().padStart(2, '0')}`;
   102→   102→}

   104→   104→function formatRelativeDate(dateStr: string): string {
   105→   105→  const diff = Date.now() - new Date(dateStr).getTime();
   106→   106→  const min = Math.floor(diff / 60000);
   107→   107→  if (min < 1) return "À l'instant";
   108→   108→  if (min < 60) return `Il y a ${min}min`;
   109→   109→  const h = Math.floor(min / 60);
   110→   110→  if (h < 24) return `Il y a ${h}h`;
   111→   111→  const d = Math.floor(h / 24);
   112→   112→  if (d < 7) return `Il y a ${d}j`;
   113→   113→  return new Date(dateStr).toLocaleDateString('fr-FR', { day: 'numeric', month: 'short' });
   114→   114→}

   116→   116→// ─── Segment generator for movies ───
   117→   117→function generateMovieSegments(runtime: number): { index: number; title: string; start: number }[] {
   118→   118→  if (!runtime || runtime <= 0) return [{ index: 0, title: 'Début', start: 0 }];
   119→   119→  const segmentLength = 5;
   120→   120→  const segments = [];
   121→   121→  for (let i = 0; i < Math.ceil(runtime / segmentLength); i++) {
   122→   122→    const startMin = i * segmentLength;
   123→   123→    const endMin = Math.min((i + 1) * segmentLength, runtime);
   124→   124→    segments.push({
   125→   125→      index: i,
   126→   126→      title: `${formatTimeShort(startMin * 60)} - ${formatTimeShort(endMin * 60)}`,
   127→   127→      start: startMin * 60,
   128→   128→    });
   129→   129→  }
   130→   130→  return segments;
   131→   131→}

   145→   145→function SmartShortPlayer({ src, title, onSwitchServer, serverIndex, totalServers }: {
   146→   146→  src: string; title: string; onSwitchServer: () => void; serverIndex: number; totalServers: number;
   147→   147→}) {
   148→   148→  const [state, setState] = useState<'loading' | 'ready' | 'error' | 'timeout'>('loading');
   149→   149→  const [failedServers, setFailedServers] = useState<Set<number>>(new Set());
   150→   150→  const [autoFallbackAttempted, setAutoFallbackAttempted] = useState(false);
   151→   151→  const [allFailed, setAllFailed] = useState(false);
   152→   152→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    setState('loading');
   156→   156→    setAutoFallbackAttempted(false);
   157→   157→    if (timerRef.current) clearTimeout(timerRef.current);
   158→   158→    timerRef.current = setTimeout(() => {
   159→   159→      setState(prev => prev === 'loading' ? 'timeout' : prev);
   160→   160→      setFailedServers(prev => new Set(prev).add(serverIndex));
   161→   161→    }, 15000);
   162→   162→    return () => { if (timerRef.current) clearTimeout(timerRef.current); };
   163→   163→  }, [src, serverIndex]);
   164→   164→
   165→   165→  // Auto-fallback when server fails
   166→   166→  useEffect(() => {
   167→   167→    if ((state === 'error' || state === 'timeout') && !autoFallbackAttempted) {
   168→   168→      setAutoFallbackAttempted(true);
   169→   169→      const newFailed = new Set(failedServers);
   170→   170→      newFailed.add(serverIndex);
   171→   171→      if (newFailed.size >= totalServers) {
   172→   172→        setAllFailed(true);
   173→   173→        return;
   174→   174→      }
   175→   175→      const timer = setTimeout(() => onSwitchServer(), 1500);
   176→   176→      return () => clearTimeout(timer);
   177→   177→    }
   178→   178→  }, [state, autoFallbackAttempted, failedServers, serverIndex, totalServers, onSwitchServer]);
   179→   179→
   180→   180→  const handleLoad = useCallback(() => {
   181→   181→    if (timerRef.current) clearTimeout(timerRef.current);
   182→   182→    setState('ready');
   183→   183→    setFailedServers(prev => { const n = new Set(prev); n.delete(serverIndex); return n; });
   184→   184→  }, [serverIndex]);
   185→   185→
   186→   186→  const handleError = useCallback(() => {
   187→   187→    if (timerRef.current) clearTimeout(timerRef.current);
   188→   188→    setState('error');
   189→   189→    setFailedServers(prev => new Set(prev).add(serverIndex));
   190→   190→  }, [serverIndex]);
   191→   191→
   192→   192→  const showOverlay = state === 'error' || state === 'timeout';
   193→   193→  const serverName = Object.values(API_CONFIG.videoServers)[serverIndex]?.name || `Serveur ${serverIndex + 1}`;
   194→   194→  const available = totalServers - failedServers.size;
   195→   195→  const isAutoFallingBack = showOverlay && !allFailed && !autoFallbackAttempted;
   196→   196→
   197→   197→  return (
   198→   198→    <div className="absolute inset-0 flex items-center justify-center">
   199→   199→      <iframe
   200→   200→        src={src}
   201→   201→        className={cn('w-full h-full transition-opacity duration-500', showOverlay ? 'opacity-0' : 'opacity-100')}
   202→   202→        style={{ aspectRatio: '16/9', minHeight: '100%', maxWidth: '177.78%' }}
   203→   203→        allowFullScreen
   204→   204→        allow="autoplay; fullscreen; picture-in-picture"
   205→   205→        referrerPolicy="origin"
   206→   206→        title={title}
   207→   207→        onLoad={handleLoad}
   208→   208→        onError={handleError}
   209→   209→      />
   210→   210→      {state === 'loading' && !showOverlay && (
   211→   211→        <div className="absolute inset-0 flex items-center justify-center z-10 bg-black/60">
   212→   212→          <div className="flex flex-col items-center gap-2">
   213→   213→            <div className="w-6 h-6 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   214→   214→            <span className="text-[10px] text-white/30">{serverName}</span>
   215→   215→          </div>
   216→   216→        </div>
   217→   217→      )}
   218→   218→      {/* Auto-fallback transition */}
   219→   219→      {isAutoFallingBack && (
   220→   220→        <div className="absolute inset-0 flex items-center justify-center z-15 bg-black/80">
   221→   221→          <div className="flex flex-col items-center gap-2 animate-pulse">
   222→   222→            <Wifi className="w-6 h-6 text-primary" />
   223→   223→            <p className="text-[11px] text-white/50">Serveur suivant...</p>
   224→   224→          </div>
   225→   225→        </div>
   226→   226→      )}
   227→   227→      {/* All servers failed */}
   228→   228→      {showOverlay && allFailed && (
   229→   229→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   230→   230→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   231→   231→            <WifiOff className="w-8 h-8 text-red-400" />
   232→   232→            <p className="text-xs font-bold text-white">Contenu indisponible</p>
   233→   233→            <p className="text-[10px] text-white/30">Aucun serveur ne peut charger ce contenu. Réessayez plus tard.</p>
   234→   234→            <button onClick={() => { setAllFailed(false); setFailedServers(new Set()); onSwitchServer(); }}
   235→   235→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   236→   236→              <RefreshCw className="w-3.5 h-3.5" />
   237→   237→              Réessayer
   238→   238→            </button>
   239→   239→          </div>
   240→   240→        </div>
   241→   241→      )}
   242→   242→      {/* Some servers still available */}
   243→   243→      {showOverlay && !allFailed && !isAutoFallingBack && (
   244→   244→        <div className="absolute inset-0 flex items-center justify-center z-20 bg-black/90">
   245→   245→          <div className="flex flex-col items-center gap-3 px-6 text-center max-w-[260px]">
   246→   246→            <AlertTriangle className="w-8 h-8 text-amber-400" />
   247→   247→            <p className="text-xs font-bold text-white">Serveur indisponible</p>
   248→   248→            <p className="text-[10px] text-white/30">{serverName} ne répond pas</p>
   249→   249→            <p className="text-[10px] text-primary/50">{available} serveur{available > 1 ? 's' : ''} restant{available > 1 ? 's' : ''}</p>
   250→   250→            <button onClick={onSwitchServer}
   251→   251→              className="flex items-center gap-1.5 px-4 py-2 rounded-full bg-primary text-black text-xs font-bold active:scale-95 transition-transform">
   252→   252→              <SkipForward className="w-3.5 h-3.5" />
   253→   253→              Serveur suivant
   254→   254→            </button>
   255→   255→            <p className="text-[9px] text-white/15">{serverIndex + 1} / {totalServers} · {failedServers.size} échoué{failedServers.size > 1 ? 's' : ''}</p>
   256→   256→          </div>
   257→   257→        </div>
   258→   258→      )}
   259→   259→    </div>
   260→   260→  );
   261→   261→}

   273→   273→// ─── Load shorts from a single endpoint, deduped ───
   274→   274→async function loadShortsFromEndpoint(
   275→   275→  endpoint: string,
   276→   276→  mediaType: 'movie' | 'tv',
   277→   277→  genreLabel?: string,
   278→   278→  maxItems: number = 8,
   279→   279→  seenIds: Set<string> = new Set()
   280→   280→): Promise<ShortItem[]> {
   281→   281→  const data = await fetchTMDB<TMDBResponse<Media>>(endpoint);
   282→   282→  if (!data?.results) return [];
   283→   283→
   284→   284→  const items: ShortItem[] = [];
   285→   285→
   286→   286→  for (const item of data.results) {
   287→   287→    if (items.length >= maxItems) break;
   288→   288→
   289→   289→    // Dedup key: same id + same mediaType = skip
   290→   290→    const dedupKey = `${item.id}-${mediaType}`;
   291→   291→    if (seenIds.has(dedupKey)) continue;
   292→   292→    seenIds.add(dedupKey);
   293→   293→
   294→   294→    if (mediaType === 'tv') {
   295→   295→      // For TV: fetch first episode
   296→   296→      const detail = await fetchTMDB<any>(`/tv/${item.id}`);
   297→   297→      if (detail?.seasons) {
   298→   298→        const firstRealSeason = detail.seasons.find((s: any) => s.season_number > 0);
   299→   299→        if (firstRealSeason) {
   300→   300→          const seasonData = await fetchTMDB<any>(`/tv/${item.id}/season/${firstRealSeason.season_number}`);
   301→   301→          if (seasonData?.episodes?.[0]) {
   302→   302→            const ep = seasonData.episodes[0];
   303→   303→            items.push({
   304→   304→              id: item.id,
   305→   305→              mediaType: 'tv',
   306→   306→              title: item.name || item.original_name || '',
   307→   307→              overview: ep.overview || item.overview || '',
   308→   308→              posterPath: item.poster_path,
   309→   309→              backdropPath: ep.still_path || item.backdrop_path,
   310→   310→              voteAverage: item.vote_average,
   311→   311→              genreLabel,
   312→   312→              season: firstRealSeason.season_number,
   313→   313→              episode: ep.episode_number,
   314→   314→              episodeName: ep.name,
   315→   315→            });
   316→   316→          }
   317→   317→        }
   318→   318→      }
   319→   319→    } else {
   320→   320→      // For movies: single card (no segment splitting for cleaner feed)
   321→   321→      items.push({
   322→   322→        id: item.id,
   323→   323→        mediaType: 'movie',
   324→   324→        title: item.title || item.original_title || '',
   325→   325→        overview: item.overview || '',
   326→   326→        posterPath: item.poster_path,
   327→   327→        backdropPath: item.backdrop_path,
   328→   328→        voteAverage: item.vote_average,
   329→   329→        genreLabel,
   330→   330→      });
   331→   331→    }
   332→   332→  }
   333→   333→
   334→   334→  return items;
   335→   335→}

   338→   338→function LikeButton({ mediaId, mediaType, segment, season, likeTriggerRef }: { mediaId: number; mediaType: string; segment: number; season?: number | null; likeTriggerRef?: React.MutableRefObject<(() => void) | null> }) {
   339→   339→  const [data, setData] = useState<LikeData>({ count: 0, liked: false });
   340→   340→  const [animating, setAnimating] = useState(false);
   341→   341→  const [justLiked, setJustLiked] = useState(false); // track if we JUST liked for animation direction
   342→   342→  const [isToggling, setIsToggling] = useState(false); // prevent double-clicks
   343→   343→  const { data: session } = useSession();
   344→   344→  const normalizedSeason = season || 0;
   345→   345→
   346→   346→  useEffect(() => {
   347→   347→    const params = new URLSearchParams({ mediaId: String(mediaId), mediaType, segment: String(segment), season: String(normalizedSeason) });
   348→   348→    fetch(`/api/shorts/like?${params}`)
   349→   349→      .then(r => r.ok ? r.json() : null)
   350→   350→      .then(d => { if (d?.count !== undefined) setData(d); })
   351→   351→      .catch(() => {});
   352→   352→  }, [mediaId, mediaType, segment, normalizedSeason]);
   353→   353→
   354→   354→  const toggleLike = async () => {
   355→   355→    if (!session || isToggling) return;
   356→   356→    setIsToggling(true);
   357→   357→    
   358→   358→    // Optimistic update — immediately show the animation
   359→   359→    const willLike = !data.liked;
   360→   360→    setJustLiked(willLike);
   361→   361→    setAnimating(true);
   362→   362→    setData(prev => ({ count: prev.count + (willLike ? 1 : -1), liked: willLike }));
   363→   363→
   364→   364→    try {
   365→   365→      const res = await fetch('/api/shorts/like', {
   366→   366→        method: 'POST',
   367→   367→        headers: { 'Content-Type': 'application/json' },
   368→   368→        body: JSON.stringify({ mediaId, mediaType, segment, season: normalizedSeason }),
   369→   369→      });
   370→   370→      
   371→   371→      if (!res.ok) {
   372→   372→        // HTTP error (401, 500, etc.) — revert optimistic update
   373→   373→        console.warn('[like] HTTP error:', res.status);
   374→   374→        setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   375→   375→      } else {
   376→   376→        const d = await res.json();
   377→   377→        if (d?.liked !== undefined) {
   378→   378→          // Server confirmed — use server state
   379→   379→          setData(prev => ({ count: d.count ?? prev.count, liked: d.liked }));
   380→   380→        } else if (d?.error) {
   381→   381→          // Server error — revert optimistic update
   382→   382→          console.warn('[like] API error:', d.error);
   383→   383→          setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   384→   384→        }
   385→   385→      }
   386→   386→    } catch (err) {
   387→   387→      // Network error — revert optimistic update
   388→   388→      console.warn('[like] Network error:', err);
   389→   389→      setData(prev => ({ count: prev.count + (willLike ? -1 : 1), liked: !willLike }));
   390→   390→    }
   391→   391→    setTimeout(() => { setAnimating(false); setJustLiked(false); setIsToggling(false); }, 600);
   392→   392→  };
   393→   393→
   394→   394→  // Expose toggleLike via ref so parent (double-tap) can trigger it
   395→   395→  useEffect(() => {
   396→   396→    if (likeTriggerRef) likeTriggerRef.current = toggleLike;
   397→   397→  }); // update every render to keep ref fresh
   398→   398→
   399→   399→  return (
   400→   400→    <button onClick={toggleLike} className="flex flex-col items-center gap-0.5 group active:scale-90 transition-transform">
   401→   401→      <div className="relative">
   402→   402→        <Heart
   403→   403→          className={cn(
   404→   404→            'w-7 h-7 transition-all duration-300',
   405→   405→            animating && justLiked && 'scale-125',
   406→   406→            data.liked
   407→   407→              ? 'text-red-500 fill-red-500 drop-shadow-[0_0_8px_rgba(239,68,68,0.4)]'
   408→   408→              : 'text-white/80 group-hover:text-white'
   409→   409→          )}
   410→   410→          strokeWidth={data.liked ? 0 : 1.5}
   411→   411→        />
   412→   412→        {animating && (
   413→   413→          <div className="absolute inset-0 flex items-center justify-center">
   414→   414→            <div className={cn(
   415→   415→              'w-12 h-12 rounded-full animate-ping-once',
   416→   416→              justLiked ? 'bg-red-500/25' : 'bg-white/10'
   417→   417→            )} />
   418→   418→          </div>

   814→   814→  switch (profileType) {

   822→   822→  }
   823→   823→}

   825→   825→// ─── Helper: add page param to endpoint ───
