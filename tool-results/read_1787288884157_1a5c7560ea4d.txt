    51→// Fetch helper
    52→const fetchTMDB = async <T,>(endpoint: string): Promise<T | null> => {
    53→  try {
    54→    const res = await fetch(
    55→      `${API_CONFIG.tmdb.baseUrl}${endpoint}${endpoint.includes('?') ? '&' : '?'}api_key=${API_CONFIG.tmdb.apiKey}&language=fr-FR`
    56→    );
    57→    return res.ok ? await res.json() : null;
    58→  } catch {
    59→    return null;
    60→  }
    61→};

    63→// Promotional banner
    64→function PromoBanner() {
    65→  const [items, setItems] = useState<Media[]>([]);
    66→  const [currentIndex, setCurrentIndex] = useState(0);
    67→
    68→  useEffect(() => {
    69→    const load = async () => {
    70→      const data = await fetchTMDB<TMDBResponse<Media>>('/movie/now_playing?region=FR');
    71→      if (data?.results) {
    72→        setItems(data.results.filter(m => m.backdrop_path).slice(0, 5));
    73→      }
    74→    };
    75→    load();
    76→  }, []);
    77→
    78→  useEffect(() => {
    79→    if (items.length <= 1) return;
    80→    const interval = setInterval(() => {
    81→      setCurrentIndex(prev => (prev + 1) % items.length);
    82→    }, 6000);
    83→    return () => clearInterval(interval);
    84→  }, [items.length]);
    85→
    86→  if (items.length === 0) return null;
    87→
    88→  const current = items[currentIndex];
    89→  const title = current.title || current.name || '';
    90→
    91→  return (
    92→    <div className="relative w-full h-[35vh] sm:h-[40vh] overflow-hidden">
    93→      <div className="absolute inset-0">
    94→        <img
    95→          src={`${API_CONFIG.tmdb.imageUrl}/original${current.backdrop_path}`}
    96→          alt={title}
    97→          className="w-full h-full object-cover transition-all duration-1000"
    98→          key={currentIndex}
    99→        />
   100→      </div>
   101→      <div className="absolute inset-0 bg-gradient-to-b from-background/30 via-background/10 to-background" />
   102→      <div className="absolute inset-0 bg-gradient-to-r from-background/50 via-transparent to-transparent" />
   103→
   104→      {/* Dots */}
   105→      {items.length > 1 && (
   106→        <div className="absolute bottom-3 right-6 flex items-center gap-1.5">
   107→          {items.map((_, i) => (
   108→            <button
   109→              key={i}
   110→              onClick={() => setCurrentIndex(i)}
   111→              className={`w-1.5 h-1.5 rounded-full transition-all duration-300 ${
   112→                i === currentIndex ? 'bg-primary w-4' : 'bg-white/30 hover:bg-white/50'
   113→              }`}
   114→            />
   115→          ))}
   116→        </div>
   117→      )}
   118→    </div>
   119→  );
   120→}

   467→        <h1 className="text-2xl sm:text-3xl md:text-4xl font-bold text-foreground mb-1 animate-fade-in">
   468→          {managing ? 'Gérer les profils' : 'Qui regarde ?'}
   469→        </h1>
   470→        {managing && (
   471→          <p className="text-sm text-primary mb-2 animate-fade-in">
   472→            Cliquez sur un profil pour le modifier
   473→          </p>
   474→        )}
   475→
   476→        {/* Profile cards */}
   477→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-6 sm:gap-8 md:gap-10 justify-items-center my-8 sm:my-10">
   478→          {profiles.map((profile, i) => {
   479→            const imgSrc = PROFILE_IMAGES[profile.type];
   480→            const gradient = PROFILE_COLORS[profile.type];
   481→            const desc = PROFILE_DESCRIPTIONS[profile.type];
   482→            const isActive = currentProfile?.type === profile.type;
   483→            const [hovered, setHovered] = useState(false);
   484→            const [visible, setVisible] = useState(false);
   485→            const [imgLoaded, setImgLoaded] = useState(false);
   486→
   487→            useEffect(() => {
   488→              const timer = setTimeout(() => setVisible(true), 150 + i * 120);
   489→              return () => clearTimeout(timer);
   490→            }, []);
   491→
   492→            return (
   493→              <button
   494→                key={profile.type}
   495→                onClick={() => handleSelect(profile)}
   496→                onMouseEnter={() => setHovered(true)}
   497→                onMouseLeave={() => setHovered(false)}
   498→                className={`flex flex-col items-center gap-3 group transition-all duration-500 ${
   499→                  visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-6'
   500→                }`}
   501→              >
   502→                {/* Avatar */}
   503→                <div
   504→                  className={`relative w-28 h-28 sm:w-32 sm:h-32 md:w-36 md:h-36 rounded-xl overflow-hidden
   505→                    ring-2 ${isActive ? 'ring-primary ring-offset-2 ring-offset-background' : hovered ? 'ring-white/60 ring-offset-2 ring-offset-background' : 'ring-transparent'}
   506→                    transition-all duration-300 ${hovered ? 'scale-105' : 'scale-100'}
   507→                    shadow-lg bg-muted`}
   508→                >
   509→                  {!imgLoaded && <div className="absolute inset-0 bg-muted animate-pulse" />}
   510→                  <img
   511→                    src={imgSrc}
   512→                    alt={profile.name}
   513→                    className={`w-full h-full object-cover transition-all duration-300 ${
   514→                      imgLoaded ? 'opacity-100' : 'opacity-0'
   515→                    } group-hover:scale-110`}
   516→                    onLoad={() => setImgLoaded(true)}
   517→                  />
   518→
   519→                  {/* Active badge */}
   520→                  {isActive && !managing && (
   521→                    <div className="absolute bottom-1 right-1 w-6 h-6 rounded-full bg-primary flex items-center justify-center shadow-md">
   522→                      <Check className="w-3.5 h-3.5 text-black" />
   523→                    </div>
   524→                  )}
   525→
   526→                  {/* Managing overlay */}
   527→                  {managing && (
   528→                    <div className={`absolute inset-0 bg-black/50 flex items-center justify-center transition-opacity duration-200 ${
   529→                      hovered ? 'opacity-100' : 'opacity-0'
   530→                    }`}>
   531→                      <Pencil className="w-8 h-8 text-white" />
   532→                    </div>
   533→                  )}
   534→
   535→                  {/* Hover overlay */}
   536→                  {!managing && !isActive && hovered && (
   537→                    <div className="absolute inset-0 bg-black/30 flex items-center justify-center animate-fade-in">
   538→                      <div className="w-12 h-12 rounded-full bg-white/20 backdrop-blur-sm flex items-center justify-center">
   539→                        <Check className="w-7 h-7 text-white" />
   540→                      </div>
   541→                    </div>
   542→                  )}
   543→                </div>
   544→
   545→                {/* Name */}
   546→                <span className={`text-sm sm:text-base font-semibold transition-colors duration-200 ${
   547→                  isActive ? 'text-primary' : hovered ? 'text-foreground' : 'text-foreground/60'
   548→                }`}>
   549→                  {profile.name}
   550→                </span>
   551→
   552→                {/* Description */}
   553→                <span className="text-[11px] text-muted-foreground/40 max-w-[120px] text-center">
   554→                  {desc}
   555→                </span>
   556→              </button>
   557→            );
   558→          })}
   559→        </div>
   560→
   561→        {/* Managing toggle */}
   562→        {managing && (
   563→          <button
   564→            onClick={() => setManaging(false)}
   565→            className="flex items-center gap-2 px-6 py-2.5 rounded-full bg-primary text-black font-semibold hover:bg-primary/90 transition-all text-sm"
   566→          >
   567→            Terminé
   568→          </button>
   569→        )}
   570→      </div>
   571→
   572→      {/* Bottom */}
   573→      <div className="py-3 text-center">
   574→        <p className="text-[10px] text-muted-foreground/20">
   575→          © 2026 Netplus · Aeronlabs + IAgen
   576→        </p>
   577→      </div>
   578→    </div>
   579→  );
   580→}
