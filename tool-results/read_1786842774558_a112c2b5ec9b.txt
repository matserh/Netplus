    33→function ConfettiEffect() {
    34→  const [particles, setParticles] = useState<Array<{ id: number; x: number; y: number; color: string; delay: number; size: number }>>([]);
    35→
    36→  useEffect(() => {
    37→    const colors = ['#FFD700', '#FFA500', '#FF6347', '#FF4500', '#FFD700', '#DAA520'];
    38→    const p = Array.from({ length: 40 }, (_, i) => ({
    39→      id: i,
    40→      x: Math.random() * 100,
    41→      y: -10 - Math.random() * 20,
    42→      color: colors[Math.floor(Math.random() * colors.length)],
    43→      delay: Math.random() * 2,
    44→      size: 4 + Math.random() * 8,
    45→    }));
    46→    setParticles(p);
    47→  }, []);
    48→
    49→  if (particles.length === 0) return null;
    50→
    51→  return (
    52→    <div className="fixed inset-0 pointer-events-none z-50 overflow-hidden">
    53→      {particles.map(p => (
    54→        <div
    55→          key={p.id}
    56→          className="absolute animate-confetti-fall"
    57→          style={{
    58→            left: `${p.x}%`,
    59→            top: `${p.y}%`,
    60→            width: p.size,
    61→            height: p.size,
    62→            backgroundColor: p.color,
    63→            borderRadius: Math.random() > 0.5 ? '50%' : '2px',
    64→            animationDelay: `${p.delay}s`,
    65→          }}
    66→        />
    67→      ))}
    68→    </div>
    69→  );
    70→}
